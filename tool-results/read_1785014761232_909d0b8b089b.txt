   127→function mapUrlToPage(url: string): { page: PageName; params?: Record<string, unknown> } {
   128→  const clean = url.replace(/^\/#?/, '').replace(/\/$/, '')
   129→  const pageMap: Record<string, PageName> = {
   130→    '': 'home',
   131→    home: 'home',
   132→    products: 'products',
   133→    categories: 'products',
   134→    offers: 'products',
   135→    about: 'home',
   136→    contact: 'contact',
   137→    cart: 'cart',
   138→    checkout: 'checkout',
   139→    login: 'login',
   140→    register: 'register',
   141→    wishlist: 'wishlist',
   142→    profile: 'profile',
   143→    orders: 'orders',
   144→    faq: 'contact',
   145→    'quality-promise': 'home',
   146→    'delivery-policy': 'delivery-policy',
   147→    'return-policy': 'return-policy',
   148→    terms: 'terms',
   149→    privacy: 'privacy',
   150→    compare: 'products',
   151→  }
   152→
   153→  if (clean.startsWith('product/')) {
   154→    return { page: 'product-detail', params: { slug: clean.split('/')[1] } }
   155→  }
   156→  if (clean.startsWith('category/')) {
   157→    return { page: 'products', params: { category: clean.split('/')[1] } }
   158→  }
   159→  if (clean.startsWith('order/')) {
   160→    return { page: 'order-detail', params: { id: clean.split('/')[1] } }
   161→  }
   162→
   163→  return { page: pageMap[clean] || 'home' }
   164→}
