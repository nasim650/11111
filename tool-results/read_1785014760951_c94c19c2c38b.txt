    66→
    67→function OrderTimeline({ status }: { status: string }) {
    68→  const currentIdx = statusSteps.indexOf(status)
    69→  const isCancelled = status === 'cancelled'
    70→
    71→  return (
    72→    <div className="flex items-center gap-0 my-3">
    73→      {statusSteps.map((step, idx) => {
    74→        const isCompleted = !isCancelled && idx <= currentIdx
    75→        const isCurrent = !isCancelled && idx === currentIdx
    76→        const StatusIcon = statusConfig[step]?.icon || Package
    77→
    78→        return (
    79→          <div key={step} className="flex items-center flex-1">
    80→            <div className="flex flex-col items-center flex-1">
    81→              <div
    82→                className={`h-7 w-7 rounded-full flex items-center justify-center transition-all duration-300 ${
    83→                  isCancelled
    84→                    ? 'bg-red-100 text-red-400'
    85→                    : isCompleted
    86→                    ? 'bg-gradient-to-br from-emerald-500 to-emerald-700 text-white shadow-sm shadow-emerald-200'
    87→                    : 'bg-gray-100 text-gray-400 border border-gray-200'
    88→                } ${isCurrent ? 'ring-2 ring-emerald-600/20 ring-offset-1' : ''}`}
    89→              >
    90→                {isCompleted ? (
    91→                  <CheckCircle2 className="h-3.5 w-3.5" />
    92→                ) : (
    93→                  <StatusIcon className="h-3 w-3" />
    94→                )}
    95→              </div>
    96→              <span className={`text-[9px] mt-1 font-medium ${isCompleted ? 'text-emerald-700' : 'text-gray-400'}`}>
    97→                {statusConfig[step]?.label || step}
    98→              </span>
    99→            </div>
   100→            {idx < statusSteps.length - 1 && (
   101→              <div className={`h-0.5 w-full -mt-5 rounded-full ${
   102→                !isCancelled && idx < currentIdx ? 'bg-emerald-500' : 'bg-gray-200'
   103→              }`} />
   104→            )}
   105→          </div>
   106→        )
   107→      })}
   108→    </div>
   109→  )
   110→}

   112→function getEstimatedDelivery(createdAt: string, status: string): string | null {
   113→  if (status === 'delivered' || status === 'cancelled') return null
   114→  const created = new Date(createdAt)
   115→  const est = new Date(created.getTime() + 5 * 24 * 60 * 60 * 1000)
   116→  return est.toLocaleDateString('en-BD', { month: 'short', day: 'numeric' })
   117→}

   119→function formatPaymentMethod(method: string): string {
   120→  const map: Record<string, string> = {
   121→    cod: 'Cash on Delivery',
   122→    online: 'Online Payment',
   123→  }
   124→  return map[method] || method.charAt(0).toUpperCase() + method.slice(1)
   125→}

   191→export default function OrdersPage() {
   192→  const { navigate, isAuthenticated, user, addToCart } = useAppStore()
   193→  const [orders, setOrders] = useState<Order[]>([])
   194→  const [loading, setLoading] = useState(true)
   195→  const [searchQuery, setSearchQuery] = useState('')
   196→  const [activeTab, setActiveTab] = useState('all')
   197→  const [currentPage, setCurrentPage] = useState(1)
   198→  const ordersPerPage = 5
   199→
   200→  useEffect(() => {
   201→    if (!isAuthenticated) {
   202→      navigate('login')
   203→      return
   204→    }
   205→    if (user?.id) {
   206→      fetch(`/api/orders?userId=${user.id}`)
   207→        .then((r) => r.json())
   208→        .then((data) => {
   209→          const ords = extractItems<Order>(data)
   210→          setOrders(ords)
   211→        })
   212→        .catch(() => {})
   213→        .finally(() => setLoading(false))
   214→    }
   215→  }, [isAuthenticated, user?.id, navigate])
   216→
   217→  const filteredOrders = useMemo(() => {
   218→    let filtered = orders
   219→
   220→    if (activeTab !== 'all') {
   221→      filtered = filtered.filter((o) => o.orderStatus === activeTab)
   222→    }
   223→
   224→    if (searchQuery.trim()) {
   225→      const q = searchQuery.toLowerCase()
   226→      filtered = filtered.filter(
   227→        (o) =>
   228→          o.orderNumber.toLowerCase().includes(q) ||
   229→          o.items?.some((i) => i.productName.toLowerCase().includes(q))
   230→      )
   231→    }
   232→
   233→    return filtered
   234→  }, [orders, activeTab, searchQuery])
   235→
   236→  // Pagination
   237→  const totalPages = Math.ceil(filteredOrders.length / ordersPerPage)
   238→  const paginatedOrders = filteredOrders.slice(
   239→    (currentPage - 1) * ordersPerPage,
   240→    currentPage * ordersPerPage
   241→  )
   242→
   243→  useEffect(() => {
   244→    setCurrentPage(1)
   245→  }, [activeTab, searchQuery])
   246→
   247→  const handleReorder = async (order: Order) => {
   248→    if (!order.items?.length) return
   249→    let added = 0
   250→    for (const item of order.items) {
   251→      try {
   252→        const cartItem: Omit<CartItem, 'id'> = {
   253→          productId: `reorder-${Date.now()}`,
   254→          productName: item.productName,
   255→          price: item.price,
   256→          quantity: item.quantity,
   257→          unit: 'kg',
   258→          image: item.image || '/placeholder.png',
   259→          stock: 999,
   260→        }
   261→        await addToCart(cartItem)
   262→        added++
   263→      } catch {
   264→        // continue
   265→      }
   266→    }
   267→    toast.success(`${added} item${added !== 1 ? 's' : ''} added to cart`)
   268→  }
   269→
   270→  const handleTrack = (orderId: string) => {
   271→    navigate('order-detail', { id: orderId })
   272→  }
   273→
   274→  // ─── Order Status Counts ──────────────────────────────────────────────────
   275→
   276→  const statusCounts = useMemo(() => {
   277→    const counts: Record<string, number> = { all: orders.length }
   278→    orders.forEach((o) => {
   279→      counts[o.orderStatus] = (counts[o.orderStatus] || 0) + 1
   280→    })
   281→    return counts
   282→  }, [orders])
   283→
   284→  if (!isAuthenticated) return null
   285→
   286→  const tabs = [
   287→    { value: 'all', label: 'All' },
   288→    { value: 'pending', label: 'Pending' },
   289→    { value: 'processing', label: 'Processing' },
   290→    { value: 'shipped', label: 'Shipped' },
   291→    { value: 'delivered', label: 'Delivered' },
   292→    { value: 'cancelled', label: 'Cancelled' },
   293→  ]
   294→
   295→  // ─── Pagination with ellipsis ─────────────────────────────────────────────
   296→
   297→  const renderPagination = () => {
   298→    if (totalPages <= 1) return null
   299→
   300→    const pages: (number | string)[] = []
   301→    if (totalPages <= 7) {
   302→      for (let i = 1; i <= totalPages; i++) pages.push(i)
   303→    } else {
   304→      pages.push(1)
   305→      if (currentPage > 3) pages.push('...')
   306→      for (let i = Math.max(2, currentPage - 1); i <= Math.min(totalPages - 1, currentPage + 1); i++) {
   307→        pages.push(i)
   308→      }
   309→      if (currentPage < totalPages - 2) pages.push('...')
   310→      pages.push(totalPages)
   311→    }
   312→
   313→    return (
   314→      <div className="flex items-center justify-center gap-1.5 pt-8">
   315→        <Button
   316→          variant="outline"
   317→          size="sm"
   318→          className="h-9 w-9 p-0 rounded-lg"
   319→          disabled={currentPage === 1}
   320→          onClick={() => setCurrentPage((p) => Math.max(1, p - 1))}
   321→        >
   322→          <ChevronLeft className="h-4 w-4" />
   323→        </Button>
   324→        {pages.map((page, idx) =>
   325→          typeof page === 'string' ? (
   326→            <span key={`ellipsis-${idx}`} className="px-2 text-gray-400">
   327→              <MoreHorizontal className="h-4 w-4" />
   328→            </span>
   329→          ) : (
   330→            <Button
   331→              key={page}
   332→              variant={page === currentPage ? 'default' : 'outline'}
   333→              size="sm"
   334→              className={`h-9 w-9 p-0 rounded-lg font-semibold ${page === currentPage ? 'bg-emerald-600 hover:bg-emerald-700 shadow-sm' : ''}`}
   335→              onClick={() => setCurrentPage(page)}
   336→            >
   337→              {page}
   338→            </Button>
   339→          )
   340→        )}
   341→        <Button
   342→          variant="outline"
   343→          size="sm"
   344→          className="h-9 w-9 p-0 rounded-lg"
   345→          disabled={currentPage === totalPages}
   346→          onClick={() => setCurrentPage((p) => Math.min(totalPages, p + 1))}
   347→        >
   348→          <ChevronRightIcon className="h-4 w-4" />
   349→        </Button>
   350→      </div>
   351→    )
   352→  }
   353→
   354→  return (
   355→    <div className="min-h-screen bg-gradient-to-b from-gray-50 to-white">
   356→      {/* ─── Header ────────────────────────────────────────────────────────── */}
   357→      <div className="bg-gradient-to-r from-emerald-700 via-emerald-600 to-teal-600 py-8">
   358→        <div className="container mx-auto px-4 max-w-4xl">
   359→          <div className="flex items-center gap-3 mb-2">
   360→            <div className="h-12 w-12 rounded-xl bg-white/15 flex items-center justify-center backdrop-blur-sm">
   361→              <Package className="h-6 w-6 text-white" />
   362→            </div>
   363→            <div>
   364→              <h1 className="text-2xl font-bold text-white">My Orders</h1>
   365→              <p className="text-sm text-emerald-100 mt-0.5">Track, manage and reorder your purchases</p>
   366→            </div>
   367→          </div>
   368→
   369→          {/* Quick Stats */}
   370→          <div className="grid grid-cols-3 gap-3 mt-6">
   371→            {[
   372→              { label: 'Total Orders', value: orders.length, icon: Package, color: 'bg-white/15' },
   373→              { label: 'Active', value: orders.filter((o) => ['pending', 'confirmed', 'processing', 'shipped'].includes(o.orderStatus)).length, icon: Clock, color: 'bg-amber-400/30' },
   374→              { label: 'Delivered', value: orders.filter((o) => o.orderStatus === 'delivered').length, icon: PackageCheck, color: 'bg-emerald-400/30' },
   375→            ].map((stat) => (
   376→              <div key={stat.label} className={`${stat.color} rounded-xl px-4 py-3 backdrop-blur-sm flex items-center gap-3`}>
   377→                <stat.icon className="h-5 w-5 text-white/80" />

   381→                </div>

   383→            ))}

   385→        </div>
   386→      </div>
   387→
   388→      <div className="container mx-auto px-4 py-6 max-w-4xl">
   389→        {/* ─── Search + Tabs ──────────────────────────────────────────────── */}
   390→        <div className="space-y-4 mb-6">
   391→          <div className="flex gap-2">
   392→            <div className="relative flex-1">
   393→              <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   394→              <Input
   395→                placeholder="Search by order number or product name..."
   396→                value={searchQuery}
   397→                onChange={(e) => setSearchQuery(e.target.value)}
   398→                className="pl-10 h-11 bg-white border-gray-200 shadow-sm rounded-xl focus:border-emerald-400 focus:ring-emerald-200"
   399→              />

   401→            <Button
   402→              variant="outline"
   403→              size="icon"

   405→              onClick={() => {
   406→                if (user?.id) {
   407→                  setLoading(true)
   408→                  fetch(`/api/orders?userId=${user.id}`)
   409→                    .then((r) => r.json())
   410→                    .then((data) => {
   411→                      const ords = extractItems<Order>(data)
   412→                      setOrders(ords)
   413→                    })
   414→                    .catch(() => {})
   415→                    .finally(() => setLoading(false))

   419→              <RefreshCw className="h-4 w-4" />
   420→            </Button>
   421→          </div>
   422→
   423→          <Tabs value={activeTab} onValueChange={setActiveTab}>

   430→                const TabIcon = tabConfig?.icon

   452→        {/* ─── Loading State ──────────────────────────────────────────────── */}

   471→            <AnimatePresence mode="popLayout">

   491→                        {/* Order header with status */}
