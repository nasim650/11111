   216→// ── Flash Sale Countdown ──
   217→function FlashSaleCountdown() {
   218→  const [timeLeft, setTimeLeft] = useState({ hours: 0, minutes: 0, seconds: 0 })
   219→
   220→  useEffect(() => {
   221→    const getTargetTime = () => {
   222→      const now = new Date()
   223→      const end = new Date(now)
   224→      end.setHours(23, 59, 59, 999)
   225→      return end.getTime()
   226→    }
   227→
   228→    const updateCountdown = () => {
   229→      const target = getTargetTime()
   230→      const now = Date.now()
   231→      const diff = Math.max(target - now, 0)
   232→
   233→      setTimeLeft({
   234→        hours: Math.floor(diff / (1000 * 60 * 60)),
   235→        minutes: Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)),
   236→        seconds: Math.floor((diff % (1000 * 60)) / 1000),
   237→      })
   238→    }
   239→
   240→    updateCountdown()
   241→    const timer = setInterval(updateCountdown, 1000)
   242→    return () => clearInterval(timer)
   243→  }, [])
   244→
   245→  const pad = (n: number) => n.toString().padStart(2, '0')
   246→  const isEndingSoon = timeLeft.hours === 0 && timeLeft.minutes < 60
   247→
   248→  return (
   249→    <div className="flex items-center gap-2">
   250→      {isEndingSoon && (
   251→        <span
   252→          className="text-[10px] font-medium bg-white/15 text-white px-2 py-0.5 rounded-full mr-1"
   253→        >
   254→          Ending soon
   255→        </span>
   256→      )}
   257→      <div className="flex items-center gap-1">
   258→        <FlipDigit value={pad(timeLeft.hours)} label="ঘণ্টা" />
   259→        <span className="text-white font-bold text-xl mx-0.5 animate-pulse">:</span>
   260→        <FlipDigit value={pad(timeLeft.minutes)} label="মিনিট" />
   261→        <span className="text-white font-bold text-xl mx-0.5 animate-pulse">:</span>
   262→        <FlipDigit value={pad(timeLeft.seconds)} label="সেকেন্ড" />
   263→      </div>
   264→    </div>
   265→  )
   266→}

   369→// ── Hero Banner Carousel ──
   370→function HeroBannerCarousel({ banners, navigate }: {
   371→  banners: Banner[]
   372→  navigate: (page: PageName, params?: Record<string, unknown>) => void
   373→}) {
   374→  const [currentBanner, setCurrentBanner] = useState(0)
   375→  const [isPaused, setIsPaused] = useState(false)
   376→  const [progress, setProgress] = useState(0)
   377→  const [scrollY, setScrollY] = useState(0)
   378→  const intervalRef = useRef<NodeJS.Timeout | null>(null)
   379→
   380→  // Parallax scroll tracking
   381→  useEffect(() => {
   382→    const handleScroll = () => {
   383→      setScrollY(window.scrollY)
   384→    }
   385→    window.addEventListener('scroll', handleScroll, { passive: true })
   386→    return () => window.removeEventListener('scroll', handleScroll)
   387→  }, [])
   388→
   389→  // Auto-play with progress
   390→  useEffect(() => {
   391→    if (banners.length <= 1 || isPaused) return
   392→
   393→    const SLIDE_DURATION = 5000
   394→    const TICK = 50
   395→    let elapsed = 0
   396→
   397→    intervalRef.current = setInterval(() => {
   398→      elapsed += TICK
   399→      setProgress((elapsed / SLIDE_DURATION) * 100)
   400→
   401→      if (elapsed >= SLIDE_DURATION) {
   402→        elapsed = 0
   403→        setCurrentBanner((prev) => (prev + 1) % banners.length)
   404→        setProgress(0)
   405→      }
   406→    }, TICK)
   407→
   408→    return () => {
   409→      if (intervalRef.current) clearInterval(intervalRef.current)
   410→    }
   411→  }, [banners.length, isPaused, currentBanner])
   412→
   413→  const handlePrev = () => {
   414→    setCurrentBanner((prev) => (prev - 1 + banners.length) % banners.length)
   415→    setProgress(0)
   416→  }
   417→
   418→  const handleNext = () => {
   419→    setCurrentBanner((prev) => (prev + 1) % banners.length)
   420→    setProgress(0)
   421→  }
   422→
   423→  const handleDotClick = (idx: number) => {
   424→    setCurrentBanner(idx)
   425→    setProgress(0)
   426→  }
   427→
   428→  if (banners.length === 0) return null
   429→
   430→  // Parallax offset - capped at 60px
   431→  const parallaxOffset = Math.min(scrollY * 0.3, 60)
   432→
   433→  return (
   434→    <div
   435→      className="relative h-64 sm:h-80 md:h-[440px] lg:h-[520px] overflow-hidden bg-muted"
   436→      onMouseEnter={() => setIsPaused(true)}
   437→      onMouseLeave={() => setIsPaused(false)}
   438→    >
   439→      <AnimatePresence mode="wait">
   440→        <motion.div
   441→          key={currentBanner}
   442→          initial={{ opacity: 0, x: 60 }}
   443→          animate={{ opacity: 1, x: 0 }}
   444→          exit={{ opacity: 0, x: -60 }}
   445→          transition={{ duration: 0.5, ease: 'easeInOut' }}
   446→          className="absolute inset-0"
   447→        >
   448→          {banners[currentBanner].desktopImage ? (
   449→            <>
   450→              {/* Parallax image container */}
   451→              <div
   452→                className="absolute inset-0 w-full h-full"
   453→                style={{ transform: `translateY(${parallaxOffset}px) scale(1.1)` }}
   454→              >
   455→                <Image
   456→                  src={banners[currentBanner].desktopImage}
   457→                  alt={banners[currentBanner].title}
   458→                  fill
   459→                  className="object-cover"
   460→                  priority={currentBanner === 0}
   461→                  sizes="100vw"
   462→                />
   463→              </div>
   464→              {/* Gradient overlays - enhanced for text readability */}
   465→              <div className="absolute inset-0 bg-gradient-to-r from-black/80 via-black/50 to-black/20" />
   466→              <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-black/30" />

   472→                  <motion.div
   473→                    initial={{ opacity: 0, y: 20 }}
   474→                    animate={{ opacity: 1, y: 0 }}
   475→                    transition={{ delay: 0.2, duration: 0.5 }}
   476→                  >
   477→                    <h1 className="text-2xl sm:text-3xl md:text-5xl lg:text-6xl font-extrabold text-white mb-3 leading-tight drop-shadow-lg">
   478→                      {banners[currentBanner].title}
