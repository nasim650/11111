    91→function AnimatedSection({ children, className, delay = 0, style }: {

   101→    <motion.div
   102→      ref={ref}
   103→      initial={{ opacity: 0, y: 40 }}
   104→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}

   106→      className={className}
   107→      style={style}

   136→// ── Flash Sale Countdown ──
   137→function FlashSaleCountdown() {
   138→  const [timeLeft, setTimeLeft] = useState({ hours: 0, minutes: 0, seconds: 0 })
   139→
   140→  useEffect(() => {
   141→    const getTargetTime = () => {
   142→      const now = new Date()
   143→      const end = new Date(now)
   144→      end.setHours(23, 59, 59, 999)
   145→      return end.getTime()
   146→    }
   147→
   148→    const updateCountdown = () => {
   149→      const target = getTargetTime()
   150→      const now = Date.now()
   151→      const diff = Math.max(target - now, 0)
   152→
   153→      setTimeLeft({
   154→        hours: Math.floor(diff / (1000 * 60 * 60)),
   155→        minutes: Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)),
   156→        seconds: Math.floor((diff % (1000 * 60)) / 1000),
   157→      })
   158→    }
   159→
   160→    updateCountdown()
   161→    const timer = setInterval(updateCountdown, 1000)
   162→    return () => clearInterval(timer)
   163→  }, [])
   164→
   165→  const pad = (n: number) => n.toString().padStart(2, '0')
   166→  const isEndingSoon = timeLeft.hours === 0 && timeLeft.minutes < 60
   167→
   168→  return (
   169→    <div className="flex items-center gap-2">
   170→      {isEndingSoon && (
   171→        <span
   172→          className="text-[10px] font-medium bg-white/15 text-white px-2 py-0.5 rounded-full mr-1"
   173→        >
   174→          Ending soon
   175→        </span>
   176→      )}
   177→      <div className="flex items-center gap-1">
   178→        <FlipDigit value={pad(timeLeft.hours)} label="ঘণ্টা" />
   179→        <span className="text-white font-bold text-xl mx-0.5 animate-pulse">:</span>
   180→        <FlipDigit value={pad(timeLeft.minutes)} label="মিনিট" />
   181→        <span className="text-white font-bold text-xl mx-0.5 animate-pulse">:</span>
   182→        <FlipDigit value={pad(timeLeft.seconds)} label="সেকেন্ড" />
   183→      </div>
   184→    </div>
   185→  )
   186→}

   262→// ── Hero Banner Carousel ──
   263→function HeroBannerCarousel({ banners, navigate }: {
   264→  banners: Banner[]
   265→  navigate: (page: PageName, params?: Record<string, unknown>) => void
   266→}) {
   267→  const [currentBanner, setCurrentBanner] = useState(0)
   268→  const [isPaused, setIsPaused] = useState(false)
   269→  const [progress, setProgress] = useState(0)
   270→  const [scrollY, setScrollY] = useState(0)
   271→  const intervalRef = useRef<NodeJS.Timeout | null>(null)
   272→
   273→  // Parallax scroll tracking
   274→  useEffect(() => {
   275→    const handleScroll = () => {
   276→      setScrollY(window.scrollY)
   277→    }
   278→    window.addEventListener('scroll', handleScroll, { passive: true })
   279→    return () => window.removeEventListener('scroll', handleScroll)
   280→  }, [])
   281→
   282→  // Auto-play with progress
   283→  useEffect(() => {
   284→    if (banners.length <= 1 || isPaused) return
   285→
   286→    const SLIDE_DURATION = 5000
   287→    const TICK = 50
   288→    let elapsed = 0
   289→
   290→    intervalRef.current = setInterval(() => {
   291→      elapsed += TICK
   292→      setProgress((elapsed / SLIDE_DURATION) * 100)
   293→
   294→      if (elapsed >= SLIDE_DURATION) {
   295→        elapsed = 0
   296→        setCurrentBanner((prev) => (prev + 1) % banners.length)
   297→        setProgress(0)
   298→      }
   299→    }, TICK)
   300→
   301→    return () => {
   302→      if (intervalRef.current) clearInterval(intervalRef.current)
   303→    }
   304→  }, [banners.length, isPaused, currentBanner])
   305→
   306→  const handlePrev = () => {
   307→    setCurrentBanner((prev) => (prev - 1 + banners.length) % banners.length)
   308→    setProgress(0)
   309→  }
   310→
   311→  const handleNext = () => {
   312→    setCurrentBanner((prev) => (prev + 1) % banners.length)
   313→    setProgress(0)
   314→  }
   315→
   316→  const handleDotClick = (idx: number) => {
   317→    setCurrentBanner(idx)
   318→    setProgress(0)
   319→  }
   320→
   321→  if (banners.length === 0) return null
   322→
   323→  // Parallax offset - capped at 60px
   324→  const parallaxOffset = Math.min(scrollY * 0.3, 60)
   325→
   326→  return (
   327→    <div
   328→      className="relative h-64 sm:h-80 md:h-[440px] lg:h-[520px] overflow-hidden bg-muted"
   329→      onMouseEnter={() => setIsPaused(true)}
   330→      onMouseLeave={() => setIsPaused(false)}
   331→    >
   332→      <AnimatePresence mode="wait">
   333→        <motion.div
   334→          key={currentBanner}
   335→          initial={{ opacity: 0, x: 60 }}
   336→          animate={{ opacity: 1, x: 0 }}
   337→          exit={{ opacity: 0, x: -60 }}
   338→          transition={{ duration: 0.5, ease: 'easeInOut' }}
   339→          className="absolute inset-0"
   340→        >
   341→          {banners[currentBanner].desktopImage ? (
   342→            <>
   343→              {/* Parallax image container */}
   344→              <div
   345→                className="absolute inset-0 w-full h-full"
   346→                style={{ transform: `translateY(${parallaxOffset}px) scale(1.1)` }}
   347→              >
   348→                <Image
   349→                  src={banners[currentBanner].desktopImage}
   350→                  alt={banners[currentBanner].title}
   351→                  fill
   352→                  className="object-cover"
   353→                  priority={currentBanner === 0}
   354→                  sizes="100vw"
   355→                />
   356→              </div>
   357→              {/* Gradient overlays - enhanced for text readability */}
   358→              <div className="absolute inset-0 bg-gradient-to-r from-black/80 via-black/50 to-black/20" />
   359→              <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-black/30" />
   360→              {/* Subtle emerald tint */}
   361→              <div className="absolute inset-0 bg-emerald-900/10" />
   362→              {/* Content */}
   363→              <div className="absolute inset-0 flex items-center">
   364→                <div className="p-6 md:p-12 lg:p-16 max-w-2xl">
   365→                  <motion.div
   366→                    initial={{ opacity: 0, y: 20 }}
   367→                    animate={{ opacity: 1, y: 0 }}
   368→                    transition={{ delay: 0.2, duration: 0.5 }}

   370→                    <h1 className="text-2xl sm:text-3xl md:text-5xl lg:text-6xl font-extrabold text-white mb-3 leading-tight drop-shadow-lg">
   371→                      {banners[currentBanner].title}

   511→// ── Category Section ──
   512→function CategorySection({ categories, navigate }: {
   513→  categories: Category[]
   514→  navigate: (page: PageName, params?: Record<string, unknown>) => void
   515→}) {
   516→  const scrollRef = useRef<HTMLDivElement>(null)
   517→  const [canScrollLeft, setCanScrollLeft] = useState(false)
   518→  const [canScrollRight, setCanScrollRight] = useState(true)
   519→
   520→  const checkScroll = useCallback(() => {
   521→    if (!scrollRef.current) return
   522→    const { scrollLeft, scrollWidth, clientWidth } = scrollRef.current
   523→    setCanScrollLeft(scrollLeft > 5)
   524→    setCanScrollRight(scrollLeft + clientWidth < scrollWidth - 5)
   525→  }, [])
   526→
   527→  useEffect(() => {
   528→    checkScroll()
   529→    const el = scrollRef.current
   530→    if (el) el.addEventListener('scroll', checkScroll, { passive: true })
   531→    return () => { if (el) el.removeEventListener('scroll', checkScroll) }
   532→  }, [checkScroll])
   533→
   534→  const handleScroll = (direction: 'left' | 'right') => {
   535→    if (!scrollRef.current) return
   536→    scrollRef.current.scrollBy({
