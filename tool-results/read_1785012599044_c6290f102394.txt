   127→// ─── Helper: Session ID ──────────────────────────────────────────────────────

   129→function getOrCreateSessionId(): string {
   130→  if (typeof window === 'undefined') return ''
   131→  const KEY = 'kk_session_id'
   132→  let id = localStorage.getItem(KEY)
   133→  if (!id) {
   134→    id = `session_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`
   135→    localStorage.setItem(KEY, id)
   136→  }
   137→  return id
   138→}
   139→
   140→function getSessionId(): string | null {
   141→  if (typeof window === 'undefined') return null
   142→  return localStorage.getItem('kk_session_id')
   143→}

   295→// ─── Store Implementation ─────────────────────────────────────────────────────
   296→
   297→// Module-level flag to prevent persist middleware from writing before rehydration
   298→let _storeRehydrated = false
   299→
   300→export const useAppStore = create<AppStore>()(
   301→  persist(
   302→    (set, get) => ({
   303→      // ── Navigation ────────────────────────────────────────────────────────
   304→      currentPage: 'home' as PageName,
   305→      pageParams: {},
   306→
   307→      navigate: (page, params = {}) => {
   308→        set({ currentPage: page, pageParams: params })
   309→        if (typeof window !== 'undefined') {
   310→          window.scrollTo({ top: 0, behavior: 'smooth' })
   311→        }
   312→      },
   313→
   314→      // ── Auth ──────────────────────────────────────────────────────────────
   315→      user: null,
   316→      isAuthenticated: false,
   317→      accessToken: null,
   318→
   319→      login: async (credentials) => {
   320→        const res = await fetch('/api/auth/login', {
   321→          method: 'POST',
   322→          headers: { 'Content-Type': 'application/json' },
   323→          body: JSON.stringify(credentials),
   324→        })
   325→
   326→        if (!res.ok) {
   327→          const data = await res.json()
   328→          throw new Error(data.error || 'Login failed')
   329→        }
   330→
   331→        const data = await res.json()
   332→        const user = (data.data?.user || data.user) as User
   333→        const token = data.data?.token || data.accessToken || null
   334→        set({ user, accessToken: token, isAuthenticated: true })
   335→
   336→        // Persist user and token for loadUser
   337→        // Removed manual localStorage set, Zustand persist handles this
   338→
   339→        // After login, merge guest cart into user cart and reload
   340→        await get().loadCart()
   341→        await get().loadWishlist()
   342→      },
   343→
   344→      register: async (registerData) => {
   345→        const res = await fetch('/api/auth/register', {
   346→          method: 'POST',
   347→          headers: { 'Content-Type': 'application/json' },
   348→          body: JSON.stringify(registerData),
   349→        })
   350→
   351→        if (!res.ok) {
   352→          const data = await res.json()
   353→          throw new Error(data.error || 'Registration failed')
   354→        }
   355→
   356→        const data = await res.json()
   357→        const user = (data.data?.user || data.user) as User
   358→        const token = data.data?.token || data.accessToken || null
   359→        set({ user, accessToken: token, isAuthenticated: true })
   360→
   361→        // Persist user and token for loadUser
   362→        // Removed manual localStorage set, Zustand persist handles this
   363→
   364→        // After registration, reload cart and wishlist
   365→        await get().loadCart()
   366→        await get().loadWishlist()
   367→      },
   368→
   369→      logout: () => {
   370→        set({
   371→          user: null,
   372→          accessToken: null,
   373→          isAuthenticated: false,
   374→          cartItems: [],
   375→          wishlistItems: [],
   376→        })
   377→        // Clear persisted user data but keep session id for guest cart
   378→        if (typeof window !== 'undefined') {
   379→          localStorage.removeItem('kk_user')
   380→          localStorage.removeItem('kk_token')
   381→        }
   382→      },
   383→
   384→      setUser: (user) => {
   385→        set({ user, isAuthenticated: !!user })
   386→      },
   387→
   388→      loadUser: () => {
   389→        // Zustand persist handles hydration automatically via useAppStore.persist.rehydrate()
   390→      },
   391→
   392→      // ── Cart ──────────────────────────────────────────────────────────────
   393→      cartItems: [],
   394→      cartLoading: false,
   395→
   396→      totalItems: () => {
   397→        return get().cartItems.reduce((sum, item) => sum + (Number(item.quantity) || 0), 0)
   398→      },
   399→
   400→      subtotal: () => {
   401→        return get().cartItems.reduce((sum, item) => {
   402→          const price = item.discountPrice != null ? item.discountPrice : item.price
   403→          return sum + (Number(price) || 0) * (Number(item.quantity) || 0)
   404→        }, 0)
   405→      },
   406→
   407→      addToCart: async (item) => {
   408→        const state = get()
   409→        const userId = state.user?.id
   410→        const sessionId = userId ? undefined : getOrCreateSessionId()
   411→
   412→        try {
   413→          const token = get().accessToken
   414→          const headers: Record<string, string> = { 'Content-Type': 'application/json' }
   415→          if (token) headers['Authorization'] = `Bearer ${token}`
   416→          const res = await fetch('/api/cart', {
   417→            method: 'POST',
   418→            headers,
   419→            body: JSON.stringify({
   420→              productId: item.productId,
   421→              quantity: item.quantity,
   422→              unit: item.unit,
   423→              ...(userId ? { userId } : { sessionId }),
   424→            }),
   425→          })
   426→
   427→          if (!res.ok) {
   428→            const data = await res.json()
   429→            throw new Error(data.error || 'Failed to add to cart')
   430→          }
   431→
   432→          const savedItem = await res.json()
   433→
   434→          // Update local state: if item already exists, update it; otherwise add
   435→          set((s) => {
   436→            const existingIndex = s.cartItems.findIndex(
   437→              (ci) => ci.productId === item.productId
   438→            )
   439→            if (existingIndex >= 0) {
   440→              const updated = [...s.cartItems]
   441→              updated[existingIndex] = {
   442→                ...updated[existingIndex],
   443→                id: savedItem.id,
   444→                quantity: savedItem.quantity,
   445→              }
   446→              return { cartItems: updated }
   447→            }
   448→            return {
   449→              cartItems: [
   450→                {
   451→                  id: savedItem.id,
   452→                  productId: item.productId,
   453→                  productName: item.productName,
   454→                  price: item.price,
   455→                  discountPrice: item.discountPrice,
   456→                  quantity: savedItem.quantity,
   457→                  unit: item.unit,
   458→                  image: item.image,
   459→                  stock: item.stock,
   460→                },
   461→                ...s.cartItems,
   462→              ],
   463→            }
   464→          })
   465→
   466→        } catch (error) {
   467→          console.error('Failed to add to cart:', error)
   468→          // Fallback: add locally without server sync
   469→          set((s) => {
   470→            const existingIndex = s.cartItems.findIndex(
   471→              (ci) => ci.productId === item.productId
   472→            )
   473→            if (existingIndex >= 0) {
   474→              const updated = [...s.cartItems]
   475→              updated[existingIndex] = {
   476→                ...updated[existingIndex],
   477→                quantity: updated[existingIndex].quantity + item.quantity,
   478→              }
   479→              return { cartItems: updated }
   480→            }
   481→            return {
   482→              cartItems: [
   483→                {
   484→                  ...item,
   485→                  id: `local_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   486→                  quantity: item.quantity || 1,
   487→                },
   488→                ...s.cartItems,
   489→              ],
   490→            }
   491→          })
   492→        }
   493→      },
   494→
   495→      removeFromCart: async (cartItemId) => {
   496→        // Optimistic update
   497→        const previous = get().cartItems
   498→        set((s) => ({
   499→          cartItems: s.cartItems.filter((ci) => ci.id !== cartItemId),
   500→        }))
   501→
   502→        try {
   503→          const token = get().accessToken
   504→          const delHeaders: Record<string, string> = {}
   505→          if (token) delHeaders['Authorization'] = `Bearer ${token}`
   506→          const res = await fetch(`/api/cart/${encodeURIComponent(cartItemId)}`, {
   507→            method: 'DELETE',
   508→            headers: delHeaders,
   509→          })
   510→          if (!res.ok) {
   511→            // Revert on failure
   512→            set({ cartItems: previous })
   513→            const data = await res.json()
   514→            throw new Error(data.error || 'Failed to remove item')
   515→          }
   516→        } catch (error) {
   517→          console.error('Failed to remove from cart:', error)
   518→          set({ cartItems: previous })
   519→        }
   520→      },
   521→
   522→      updateQuantity: async (cartItemId, quantity) => {
   523→        if (quantity < 1) return
   524→
   525→        const previous = get().cartItems
   526→        // Optimistic update
   527→        set((s) => ({
   528→          cartItems: s.cartItems.map((ci) =>
   529→            ci.id === cartItemId ? { ...ci, quantity } : ci
   530→          ),
   531→        }))
   532→
   533→        try {
   534→          const token = get().accessToken
   535→          const updateHeaders: Record<string, string> = { 'Content-Type': 'application/json' }
   536→          if (token) updateHeaders['Authorization'] = `Bearer ${token}`
   537→          const res = await fetch(`/api/cart/${encodeURIComponent(cartItemId)}`, {
   538→            method: 'PUT',
   539→            headers: updateHeaders,
   540→            body: JSON.stringify({ quantity }),
   541→          })
   542→          if (!res.ok) {
   543→            set({ cartItems: previous })
   544→            const data = await res.json()
   545→            throw new Error(data.error || 'Failed to update quantity')
   546→          }
   547→        } catch (error) {
   548→          console.error('Failed to update cart quantity:', error)
   549→          set({ cartItems: previous })
   550→        }
   551→      },
   552→
   553→      clearCart: async () => {
   554→        const previous = get().cartItems
   555→        set({ cartItems: [] })
   556→
   557→        // Delete all items from server
   558→        try {
   559→          const token = get().accessToken
   560→          const clearHeaders: Record<string, string> = {}
   561→          if (token) clearHeaders['Authorization'] = `Bearer ${token}`
   562→          await Promise.all(
   563→            previous.map((item) =>
   564→              fetch(`/api/cart/${encodeURIComponent(item.id)}`, {
   565→                method: 'DELETE',
   566→                headers: clearHeaders,
   567→              })
   568→            )
   569→          )
   570→        } catch (error) {
   571→          console.error('Failed to clear cart on server:', error)
   572→          // Don't revert - user explicitly cleared
   573→        }
   574→      },
   575→
   576→      loadCart: async () => {
   577→        const state = get()
   578→        const userId = state.user?.id
   579→        const sessionId = userId ? undefined : getSessionId()
   580→
   581→        if (!userId && !sessionId) return
   582→
   583→        set({ cartLoading: true })
   584→        try {
   585→          const params = new URLSearchParams()
   586→          if (userId) {
   587→            params.set('userId', userId)
   588→          } else if (sessionId) {
   589→            params.set('sessionId', sessionId)
   590→          }
   591→
   592→          const token = get().accessToken
   593→          const headers: Record<string, string> = {}
   594→          if (token) headers['Authorization'] = `Bearer ${token}`
   595→          const res = await fetch(`/api/cart?${params.toString()}`, { headers })
   596→          if (!res.ok) throw new Error('Failed to load cart')
   597→
   598→          const data = await res.json()
   599→
   600→          // Map API response to our CartItem shape
   601→          const rawItems = extractItems<Record<string, unknown>>(data)
   602→          const cartItems: CartItem[] = rawItems.map(
   603→            (item: Record<string, unknown>) => {
   604→              const product = item.product as Record<string, unknown>
   605→              const images = product?.images as Array<Record<string, unknown>> | undefined
   606→              return {
   607→                id: item.id as string,
   608→                productId: product?.id as string,
   609→                productName: product?.name as string,
   610→                price: Number(product?.price) || 0,
   611→                discountPrice: product?.discountPrice != null ? Number(product.discountPrice) : null,
   612→                quantity: item.quantity as number,
   613→                unit: (item.unit as string) || (product?.unit as string) || 'kg',
   614→                image: (images?.[0]?.url as string) || '/placeholder.png',
   615→                stock: Number(product?.stock) || 0,
   616→              }
   617→            }
   618→          )
   619→
   620→          set({ cartItems, cartLoading: false })
   621→        } catch (error) {
   622→          console.error('Failed to load cart:', error)
   623→          set({ cartLoading: false })
   624→        }
   625→      },
   626→
   627→      // ── Wishlist ──────────────────────────────────────────────────────────
   628→      wishlistItems: [],
   629→      wishlistLoading: false,
   630→
   631→      addToWishlist: async (item) => {
   632→        const state = get()
   633→        if (!state.user?.id) {
   634→          // For guests, just add locally
   635→          set((s) => ({
   636→            wishlistItems: [item, ...s.wishlistItems],
