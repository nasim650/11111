    98→    98→function StarRating({ rating, size = 14 }: { rating: number; size?: number }) {
    99→    99→  return (
   100→   100→    <div className="flex items-center gap-0.5">
   101→   101→      {Array.from({ length: 5 }).map((_, i) => {
   102→   102→        const fill = Math.min(1, Math.max(0, rating - i))
   103→   103→        if (fill >= 1) return <Star key={i} className="fill-amber-400 text-amber-400" style={{ width: size, height: size }} />
   104→   104→        if (fill >= 0.5) return <StarHalf key={i} className="fill-amber-400 text-amber-400" style={{ width: size, height: size }} />
   105→   105→        return <Star key={i} className="text-gray-200" style={{ width: size, height: size }} />
   106→   106→      })}
   107→   107→    </div>
   108→   108→  )
   109→   109→}
