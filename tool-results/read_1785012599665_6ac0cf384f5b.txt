   120→function AnimatedSection({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   121→  const ref = useRef<HTMLDivElement>(null)
   122→  const isInView = useInView(ref, { once: true, margin: '-50px' })
   123→  return (
   124→    <motion.div
   125→      ref={ref}
   126→      initial={{ opacity: 0, y: 30 }}
   127→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   128→      transition={{ duration: 0.5, delay, ease: 'easeOut' }}
   129→      className={className}
   130→    >
   131→      {children}
   132→    </motion.div>
   133→  )
   134→}

   136→// ─── Fullscreen Image Viewer ──────────────────────────────────────────────────

   138→function FullscreenViewer({
   139→  images,
   140→  initialIndex,
   141→  open,
   142→  onClose,
   143→}: {
   144→  images: Array<{ id: string; url: string; alt?: string }>
   145→  initialIndex: number
   146→  open: boolean
   147→  onClose: () => void
   148→}) {
   149→  const [idx, setIdx] = useState(initialIndex)
   150→
   151→  // Sync idx when the dialog opens or initialIndex changes (lint fix preserved)
   152→  if (open && idx !== initialIndex) {
   153→    setIdx(initialIndex)
   154→  }
   155→
   156→  useEffect(() => {
   157→    if (!open) return
   158→    const handler = (e: KeyboardEvent) => {
   159→      if (e.key === 'ArrowRight') setIdx((i) => Math.min(images.length - 1, i + 1))
   160→      if (e.key === 'ArrowLeft') setIdx((i) => Math.max(0, i - 1))
   161→      if (e.key === 'Escape') onClose()
   162→    }
   163→    window.addEventListener('keydown', handler)
   164→    return () => window.removeEventListener('keydown', handler)
   165→  }, [open, images.length, onClose])
   166→
   167→  return (
   168→    <Dialog open={open} onOpenChange={(v) => { if (!v) onClose() }}>
   169→      <DialogContent className="max-w-5xl w-full p-0 bg-black/95 border-none overflow-hidden">
   170→        <DialogTitle className="sr-only">Image Viewer</DialogTitle>
   171→        <div className="relative aspect-square w-full">
   172→          <AnimatePresence mode="wait">
   173→            <motion.div
   174→              key={idx}
   175→              initial={{ opacity: 0, scale: 0.95 }}
   176→              animate={{ opacity: 1, scale: 1 }}
   177→              exit={{ opacity: 0, scale: 0.95 }}
   178→              transition={{ duration: 0.3 }}
   179→              className="absolute inset-0"
   180→            >
   181→              <Image
   182→                src={images[idx]?.url || '/placeholder.png'}
   183→                alt={images[idx]?.alt || 'Product image'}
   184→                fill
   185→                className="object-contain"
   186→                sizes="90vw"
   187→                priority
   188→              />
   189→            </motion.div>
   190→          </AnimatePresence>
   191→          {/* Close */}
   192→          <button
   193→            onClick={onClose}
   194→            className="absolute top-4 right-4 z-10 h-10 w-10 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors"
   195→            aria-label="Close viewer"
   196→          >
   197→            <X className="h-5 w-5" />
   198→          </button>
   199→          {/* Nav arrows */}
   200→          {images.length > 1 && (
   201→            <>
   202→              <button
   203→                onClick={() => setIdx((i) => Math.max(0, i - 1))}
   204→                disabled={idx === 0}
   205→                className="absolute left-4 top-1/2 -translate-y-1/2 z-10 h-12 w-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors disabled:opacity-30"
   206→                aria-label="Previous image"
   207→              >
   208→                <ChevronLeft className="h-6 w-6" />
   209→              </button>
   210→              <button
   211→                onClick={() => setIdx((i) => Math.min(images.length - 1, i + 1))}
   212→                disabled={idx === images.length - 1}
   213→                className="absolute right-4 top-1/2 -translate-y-1/2 z-10 h-12 w-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors disabled:opacity-30"
   214→                aria-label="Next image"
   215→              >
   216→                <ChevronRight className="h-6 w-6" />
   217→              </button>
   218→            </>
   219→          )}
   220→          {/* Counter */}
   221→          <div className="absolute bottom-4 left-1/2 -translate-x-1/2 z-10 bg-black/60 backdrop-blur-sm text-white text-sm px-3 py-1.5 rounded-full">
   222→            {idx + 1} / {images.length}
   223→          </div>
   224→          {/* Thumbnails */}
   225→          {images.length > 1 && (
   226→            <div className="absolute bottom-14 left-1/2 -translate-x-1/2 z-10 flex gap-2 max-w-md overflow-x-auto px-2 py-1 scrollbar-hide">
   227→              {images.map((img, i) => (
   228→                <button
   229→                  key={img.id}
   230→                  onClick={() => setIdx(i)}
   231→                  className={`relative h-12 w-12 rounded-md overflow-hidden flex-shrink-0 border-2 transition-all ${
   232→                    i === idx ? 'border-white ring-2 ring-white/30' : 'border-white/30 opacity-60 hover:opacity-90'
   233→                  }`}
   234→                >
   235→                  <Image src={img.url} alt={img.alt || ''} fill className="object-cover" sizes="48px" />
   236→                </button>
   237→              ))}
   238→            </div>
   239→          )}
   240→        </div>
   241→      </DialogContent>
   242→    </Dialog>
   243→  )
   244→}

   248→function ScrollToTopButton() {
   249→  const [visible, setVisible] = useState(false)
   250→
   251→  useEffect(() => {
   252→    const handleScroll = () => setVisible(window.scrollY > 600)
   253→    window.addEventListener('scroll', handleScroll, { passive: true })
   254→    return () => window.removeEventListener('scroll', handleScroll)
   255→  }, [])
   256→
   257→  return (
   258→    <AnimatePresence>
   259→      {visible && (
   260→        <motion.button
   261→          initial={{ opacity: 0, scale: 0.8 }}
   262→          animate={{ opacity: 1, scale: 1 }}
   263→          exit={{ opacity: 0, scale: 0.8 }}
   264→          onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   265→          className="fixed bottom-20 md:bottom-8 right-4 z-40 h-10 w-10 rounded-full bg-emerald-600 text-white shadow-lg shadow-emerald-600/30 flex items-center justify-center hover:bg-emerald-700 transition-colors"
   266→          aria-label="Scroll to top"
   267→        >
   268→          <ArrowUp className="h-4 w-4" />
   269→        </motion.button>
   270→      )}
   271→    </AnimatePresence>
   272→  )
