    92→    92→export default function AdminOrdersPage() {
    93→    93→  const { navigate } = useAppStore()
    94→    94→  const [orders, setOrders] = useState<Order[]>([])
    95→    95→  const [loading, setLoading] = useState(true)
    96→    96→  const [statusFilter, setStatusFilter] = useState('all')
    97→    97→  const [selectedOrder, setSelectedOrder] = useState<Order | null>(null)
    98→    98→  const [detailOpen, setDetailOpen] = useState(false)
    99→    99→  const [updating, setUpdating] = useState(false)
   100→   100→
   101→   101→  // ─── Multi-select state ──────────────────────────────────────────────────
   102→   102→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   103→   103→  const [bulkPrintOpen, setBulkPrintOpen] = useState(false)
   104→   104→  const [qrDialogOpen, setQrDialogOpen] = useState(false)
   105→   105→  const [qrOrderId, setQrOrderId] = useState<string>('')
   106→   106→  const [qrOrderNumber, setQrOrderNumber] = useState<string>('')
   107→   107→
   108→   108→  // ─── Detail dialog print state ───────────────────────────────────────────
   109→   109→  // (Print loading state now lives inside the shared <OrderPrintPanel />.)
   110→   110→
   111→   111→  const fetchOrders = useCallback(async () => {
   112→   112→    try {
   113→   113→      const params = new URLSearchParams()
   114→   114→      if (statusFilter !== 'all') params.set('status', statusFilter)
   115→   115→      params.set('limit', '50')
   116→   116→      const res = await fetch(`/api/orders?${params.toString()}`)
   117→   117→      if (res.ok) {
   118→   118→        const data = await res.json()
   119→   119→        setOrders(extractItems(data))
   120→   120→      }
   121→   121→    } catch {
   122→   122→      toast.error('Failed to fetch orders')
   123→   123→    } finally {
   124→   124→      setLoading(false)
   125→   125→    }
   126→   126→  }, [statusFilter])
   127→   127→
   128→   128→  useEffect(() => { fetchOrders() }, [fetchOrders])
   129→   129→
   130→   130→  // ─── Selection helpers ────────────────────────────────────────────────────
   131→   131→
   132→   132→  const allSelected = orders.length > 0 && selectedIds.size === orders.length
   133→   133→  const someSelected = selectedIds.size > 0 && !allSelected
   134→   134→
   135→   135→  function toggleSelectAll() {
   136→   136→    if (allSelected) {
   137→   137→      setSelectedIds(new Set())
   138→   138→    } else {
   139→   139→      setSelectedIds(new Set(orders.map((o) => o.id)))
   140→   140→    }
   141→   141→  }
   142→   142→
   143→   143→  function toggleSelect(orderId: string) {
   144→   144→    setSelectedIds((prev) => {
   145→   145→      const next = new Set(prev)
   146→   146→      if (next.has(orderId)) {
   147→   147→        next.delete(orderId)
   148→   148→      } else {
   149→   149→        next.add(orderId)
   150→   150→      }
   151→   151→      return next
   152→   152→    })
   153→   153→  }
   154→   154→
   155→   155→  function clearSelection() {
   156→   156→    setSelectedIds(new Set())
   157→   157→  }
   158→   158→
   159→   159→  // ─── Open detail ─────────────────────────────────────────────────────────
   160→   160→
   161→   161→  function openDetail(order: Order) {
   162→   162→    setSelectedOrder(order)
   163→   163→    setDetailOpen(true)
   164→   164→  }
   165→   165→
   166→   166→  // ─── Update status ───────────────────────────────────────────────────────
   167→   167→
   168→   168→  async function updateStatus(orderId: string, status: string) {
   169→   169→    setUpdating(true)
   170→   170→    try {
   171→   171→      const res = await fetch(`/api/orders/${orderId}/status-history`, {
   172→   172→        method: 'POST',
   173→   173→        headers: { 'Content-Type': 'application/json' },
   174→   174→        body: JSON.stringify({ status, note: `Status updated to ${status}` }),
   175→   175→      })
   176→   176→      if (!res.ok) {
   177→   177→        const data = await res.json()
   178→   178→        throw new Error(data.error || 'Failed to update status')
   179→   179→      }
   180→   180→      toast.success(`Order status updated to ${statusLabels(status) || status}`)
   181→   181→      fetchOrders()
   182→   182→      if (selectedOrder?.id === orderId) {
   183→   183→        setSelectedOrder((prev) => prev ? { ...prev, status } : null)
   184→   184→      }
   185→   185→    } catch (error) {
   186→   186→      toast.error(error instanceof Error ? error.message : 'Failed to update order status')
   187→   187→    } finally {
   188→   188→      setUpdating(false)
   189→   189→    }
   190→   190→  }
   191→   191→
   192→   192→  // ─── Bulk status update ──────────────────────────────────────────────────
   193→   193→
   194→   194→  async function bulkUpdateStatus(newStatus: string) {
   195→   195→    if (selectedIds.size === 0) return

   198→   198→    for (const orderId of selectedIds) {
   199→   199→      try {
   200→   200→        const res = await fetch(`/api/orders/${orderId}/status-history`, {
   201→   201→          method: 'POST',
   202→   202→          headers: { 'Content-Type': 'application/json' },
   203→   203→          body: JSON.stringify({ status: newStatus, note: `Bulk status update to ${newStatus}` }),
   204→   204→        })
   205→   205→        if (res.ok) successCount++
   206→   206→      } catch {
   207→   207→        // continue
   208→   208→      }
   209→   209→    }
   210→   210→    toast.success(`${successCount} order${successCount !== 1 ? 's' : ''} updated to ${statusLabels(newStatus) || newStatus}`)
   211→   211→    clearSelection()
   212→   212→    fetchOrders()
   213→   213→    setUpdating(false)
   214→   214→  }
   215→   215→
   216→   216→  // ─── Detail dialog print helpers ─────────────────────────────────────────
   217→   217→  // (Print logic now lives in the shared <OrderPrintPanel /> component,
   218→   218→  //  which reuses the existing print API endpoints. No duplicate logic here.)

   224→   224→  }

   236→   236→    [ORDER_STATUS.PACKED]: countFor(ORDER_STATUS.PACKED),
   237→   237→    [ORDER_STATUS.READY_TO_SHIP]: countFor(ORDER_STATUS.READY_TO_SHIP),

   239→   239→    [ORDER_STATUS.DELIVERED]: countFor(ORDER_STATUS.DELIVERED),

   241→   241→    [ORDER_STATUS.RETURNED]: countFor(ORDER_STATUS.RETURNED),
   242→   242→  }

   244→   244→  return (

   253→   253→          onClick={() => navigate('admin-print-queue')}

   255→   255→          <ListChecks className="size-3.5" />
   256→   256→          Print Queue Monitor
   257→   257→        </Button>
   258→   258→      </div>
   259→   259→
   260→   260→      {/* ─── Status Tabs ──────────────────────────────────────────────────── */}
   261→   261→      <Tabs value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setLoading(true) }}>
   262→   262→        <TabsList className="flex-wrap h-auto">
   263→   263→          <TabsTrigger value="all">All ({counts.all})</TabsTrigger>
   264→   264→          {PRODUCTION_FLOW.map((s) => (
   265→   265→            <TabsTrigger key={s} value={s}>
   266→   266→              {getStatusLabel(s)} ({counts[s]})
   267→   267→            </TabsTrigger>
   268→   268→          ))}
   269→   269→          {TERMINAL_STATUSES.map((s) => (
   270→   270→            <TabsTrigger key={s} value={s}>
   271→   271→              {getStatusLabel(s)} ({counts[s]})

   273→   273→          ))}
   274→   274→        </TabsList>
   275→   275→      </Tabs>

   282→   282→          </Badge>
   283→   283→
   284→   284→          <Button
   285→   285→            size="sm"
   286→   286→            onClick={() => setBulkPrintOpen(true)}

   289→   289→            <Printer className="size-3.5" />
   290→   290→            Bulk Print
   291→   291→          </Button>
   292→   292→
   293→   293→          <DropdownMenu>
   294→   294→            <DropdownMenuTrigger asChild>
   295→   295→              <Button variant="outline" size="sm" className="gap-1.5 rounded-xl border-amber-200 text-amber-700 hover:bg-amber-50">

   297→   297→                Update Status

   301→   301→              {statusFlow.map((s) => (
   302→   302→                <DropdownMenuItem
   303→   303→                  key={s}
   304→   304→                  onClick={() => bulkUpdateStatus(s)}
   305→   305→                  disabled={updating}
   306→   306→                  className="cursor-pointer"
   307→   307→                >
   308→   308→                  <Badge variant="secondary" className={`text-[10px] mr-2 ${statusColors(s)}`}>
   309→   309→                    {s}
   310→   310→                  </Badge>
   311→   311→                  Set to {s}
   312→   312→                </DropdownMenuItem>
   313→   313→              ))}
   314→   314→            </DropdownMenuContent>
   315→   315→          </DropdownMenu>
   316→   316→
   317→   317→          <div className="flex-1" />
   318→   318→
   319→   319→          <Button
   320→   320→            variant="ghost"
   321→   321→            size="sm"
   322→   322→            onClick={clearSelection}
   323→   323→            className="gap-1.5 text-gray-500 hover:text-gray-700 rounded-xl"
   324→   324→          >
   325→   325→            <X className="size-3.5" />
   326→   326→            Clear

   331→   331→      {/* ─── Orders Table ─────────────────────────────────────────────────── */}

   353→   353→                      onCheckedChange={toggleSelectAll}

   355→   355→                    />
   356→   356→                  </TableHead>
   357→   357→                  <TableHead>Order #</TableHead>
   358→   358→                  <TableHead>Customer</TableHead>
   359→   359→                  <TableHead>Items</TableHead>
   360→   360→                  <TableHead>Total</TableHead>
   361→   361→                  <TableHead>Payment</TableHead>

   366→   366→              </TableHeader>
   367→   367→              <TableBody>

   380→   380→                    <TableCell className="font-medium text-sm">{order.orderNumber}</TableCell>
   381→   381→                    <TableCell>
   382→   382→                      <div>
   383→   383→                        <p className="text-sm">{order.customerName ?? 'N/A'}</p>
   384→   384→                        <p className="text-xs text-muted-foreground">{order.customerPhone ?? ''}</p>

   386→   386→                    </TableCell>

   397→   397→                      </div>
   398→   398→                    </TableCell>
   399→   399→                    <TableCell>

   401→   401→                        <Badge variant="secondary" className={statusColors(order.status)}>
   402→   402→                          {statusLabels(order.status) || order.status}
   403→   403→                        </Badge>
   404→   404→                        <DropdownMenu>
   405→   405→                          <DropdownMenuTrigger asChild>
   406→   406→                            <Button
   407→   407→                              variant="ghost"

   420→   420→                                disabled={updating || order.status === s}
   421→   421→                                className="cursor-pointer"
   422→   422→                              >
   423→   423→                                <Badge variant="secondary" className={`text-[10px] mr-2 ${statusColors(s)}`}>
   424→   424→                                  {statusLabels(s) || s}
   425→   425→                                </Badge>
   426→   426→                                Set to {statusLabels(s) || s}
   427→   427→                              </DropdownMenuItem>
   428→   428→                            ))}
   429→   429→                          </DropdownMenuContent>
   430→   430→                        </DropdownMenu>
   431→   431→                      </div>
   432→   432→                    </TableCell>

   440→   440→                        </Button>
   441→   441→                        <OrderPrintMenu

   446→   446→                        <Button
   447→   447→                          variant="ghost"

   464→   464→      {/* ─── Bulk Print Dialog ─────────────────────────────────────────────── */}
   465→   465→      <BulkPrintDialog
