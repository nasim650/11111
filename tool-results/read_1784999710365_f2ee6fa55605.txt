    14→function getPasswordStrength(password: string): { score: number; label: string; color: string; bgColor: string; icon: string } {
    15→  let score = 0
    16→  if (password.length >= 6) score++
    17→  if (password.length >= 8) score++
    18→  if (/[A-Z]/.test(password)) score++
    19→  if (/[0-9]/.test(password)) score++
    20→  if (/[^A-Za-z0-9]/.test(password)) score++
    21→
    22→  if (score <= 1) return { score: 20, label: 'Weak', color: 'bg-red-500', bgColor: 'bg-red-500/10', icon: '🔴' }
    23→  if (score <= 2) return { score: 40, label: 'Fair', color: 'bg-orange-500', bgColor: 'bg-orange-500/10', icon: '🟠' }
    24→  if (score <= 3) return { score: 60, label: 'Good', color: 'bg-yellow-500', bgColor: 'bg-yellow-500/10', icon: '🟡' }
    25→  if (score <= 4) return { score: 80, label: 'Strong', color: 'bg-emerald-500', bgColor: 'bg-emerald-500/10', icon: '🟢' }
    26→  return { score: 100, label: 'Very Strong', color: 'bg-emerald-600', bgColor: 'bg-emerald-600/10', icon: '💪' }
    27→}
