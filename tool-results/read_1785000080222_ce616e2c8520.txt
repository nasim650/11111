    87→  // Guest form

    89→  const [guestPhone, setGuestPhone] = useState('')
    90→  const [guestEmail, setGuestEmail] = useState('')
    91→  const [guestAddress, setGuestAddress] = useState('')

    93→  const [guestDistrict, setGuestDistrict] = useState('')
    94→
    95→  // Add address form
    96→  const [showAddAddress, setShowAddAddress] = useState(false)
    97→  const [newAddr, setNewAddr] = useState({ name: '', phone: '', address: '', city: '', district: '', postalCode: '' })
    98→
    99→  // Mobile summary sheet

   101→
   102→  useEffect(() => {
   103→    if (isAuthenticated && user?.id) {
   104→      fetch(`/api/addresses?userId=${user.id}`)

   106→        .then((data) => {
   107→          const addrs = extractItems<Address>(data)
   108→          setAddresses(addrs)

   110→          if (defaultAddr) setSelectedAddressId(defaultAddr.id)
   111→          else if (addrs.length > 0) setSelectedAddressId(addrs[0].id)
   112→        })
   113→        .catch(() => {})
   114→    }
   115→  }, [isAuthenticated, user?.id])
   116→
   117→  // Fetch loyalty balance for logged-in users
   118→  useEffect(() => {
   119→    if (isAuthenticated && user?.id) {
   120→      fetch('/api/loyalty')
   121→        .then((r) => r.json())
   122→        .then((data) => {
   123→          if (data.success) {
   124→            setLoyaltyBalance(data.data.balance || 0)
   125→          }
   126→        })
   127→        .catch(() => {})
   128→    }
   129→  }, [isAuthenticated, user?.id])
   130→
   131→  const sub = subtotal()
   132→  const standardShipping = siteConfig?.shippingCharge || 60
   133→  const expressShipping = 120
   134→  const freeThreshold = siteConfig?.freeShippingThreshold || 1500
   135→
   136→  let shipping = standardShipping
   137→  if (sub >= freeThreshold) {
   138→    shipping = 0
   139→  }

   141→    shipping = expressShipping
   142→    if (sub >= freeThreshold) {
   143→      shipping = expressShipping - standardShipping
   144→    }
   145→  }
   146→
   147→  const loyaltyDiscount = loyaltyApplied ? Math.min(loyaltyPointsToUse, Math.floor(sub * 0.2), loyaltyBalance) : 0

   149→  const effectiveLoyaltyPoints = Math.min(loyaltyPointsToUse, maxLoyaltyDiscount, loyaltyBalance)
   150→
   151→  const total = sub + shipping - discount - loyaltyDiscount
   152→  const freeShippingProgress = Math.min((sub / freeThreshold) * 100, 100)
   153→  const freeShippingRemaining = Math.max(freeThreshold - sub, 0)
   154→

   156→
   157→  const validatePhone = (p: string) => /^(?:01\d{9}|\+8801\d{9})$/.test(p)
   158→

   160→    const errors: Record<string, string> = {}
   161→    if (isAuthenticated) {
   162→      if (!selectedAddressId) errors.address = 'Please select a delivery address'
   163→    } else {
   164→      if (!guestName.trim()) errors.guestName = 'Full name is required'
   165→      if (!guestPhone.trim()) {
   166→        errors.guestPhone = 'Phone number is required'
   167→      } else if (!validatePhone(guestPhone)) {
   168→        errors.guestPhone = 'Enter a valid BD number (01XXXXXXXXX)'
   169→      }
   170→      if (!guestAddress.trim()) errors.guestAddress = 'Address is required'
   171→      if (!guestCity.trim()) errors.guestCity = 'City is required'
   172→    }
   173→    setFormErrors(errors)
   174→    return Object.keys(errors).length === 0
   175→  }
   176→
   177→  // ─── Step Navigation ─────────────────────────────────────────────────────
   178→
   179→  const goToStep = (step: number) => {
   180→    if (step <= currentStep) {
   181→      setCurrentStep(step)
   182→      return
   183→    }
   184→    if (step === 3 && currentStep === 2) {
   185→      if (!validateShippingStep()) {
   186→        toast.error('Please fill in all required fields')
   187→        return
   188→      }
   189→    }
   190→    if (step === 2 && currentStep === 1) {
   191→      // Moving from cart review to shipping — no validation needed
   192→    }
   193→    if (step > currentStep + 1) return
   194→    setCurrentStep(step)
   195→    setMaxReachedStep((prev) => Math.max(prev, step))
   196→  }
   197→
   198→  // ─── Coupon ───────────────────────────────────────────────────────────────
   199→
   200→  const handleApplyCoupon = async () => {

   202→    try {
   203→      const res = await fetch('/api/coupons/validate', {
   204→        method: 'POST',
   205→        headers: { 'Content-Type': 'application/json' },
   206→        body: JSON.stringify({ code: couponCode, totalAmount: sub }),
   207→      })
   208→      const data = await res.json()
   209→      if (!res.ok) {
   210→        toast.error(data.error || 'Invalid coupon')
   211→        return
   212→      }
   213→      const discountAmount = data.type === 'percentage'
   214→        ? Math.min(sub * (data.value / 100), data.maxDiscount || Infinity)
   215→        : data.value
   216→      setDiscount(discountAmount)
   217→      setCouponApplied(true)
   218→      toast.success('Coupon applied!')
   219→    } catch {
   220→      toast.error('Failed to validate coupon')
   221→    }
   222→  }
   223→
   224→  // ─── Add Address ──────────────────────────────────────────────────────────
   225→
   226→  const handleAddAddress = async () => {
   227→    if (!user?.id) return
   228→    try {
   229→      const res = await fetch('/api/addresses', {
   230→        method: 'POST',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({ userId: user.id, ...newAddr, isDefault: addresses.length === 0 }),
   233→      })
   234→      if (!res.ok) throw new Error('Failed to add address')
   235→      const saved = await res.json()
   236→      setAddresses((prev) => [...prev, saved])
   237→      setSelectedAddressId(saved.id)
   238→      setShowAddAddress(false)
   239→      setNewAddr({ name: '', phone: '', address: '', city: '', district: '', postalCode: '' })
   240→      toast.success('Address added')
   241→    } catch {
   242→      toast.error('Failed to add address')
   243→    }
   244→  }
   245→
   246→  // ─── Place Order ──────────────────────────────────────────────────────────
   247→
   248→  const handlePlaceOrder = async () => {
   249→    if (cartItems.length === 0) {
   250→      toast.error('Your cart is empty')
   251→      return
   252→    }
   253→
   254→    if (!validateShippingStep()) {
   255→      toast.error('Please fill in all required fields')
   256→      setCurrentStep(2)
   257→      return
   258→    }
   259→
   260→    if (!termsAccepted) {
   261→      toast.error('Please accept the terms & conditions')
   262→      return
   263→    }
   264→
   265→    setPlacing(true)
   266→    try {
   267→      const orderData: Record<string, unknown> = {
   268→        paymentMethod,
   269→        shippingMethod,
   270→        orderNote: note,
   271→        couponCode: couponCode || undefined,
   272→        discountAmount: discount,
   273→        deliveryCharge: shipping,
   274→        totalAmount: total,
   275→        loyaltyPoints: loyaltyApplied ? effectiveLoyaltyPoints : undefined,
   276→        items: cartItems.map((item) => ({
   277→          productId: item.productId,
   278→          productName: item.productName,
   279→          quantity: item.quantity,
   280→          unit: item.unit,
   281→          price: item.discountPrice != null ? item.discountPrice : item.price,
   282→          image: item.image,
   283→        })),
   284→      }
   285→
   286→      if (isAuthenticated && user?.id) {
   287→        orderData.userId = user.id
   288→        orderData.addressId = selectedAddressId
   289→      } else {
   290→        orderData.guestName = guestName
   291→        orderData.guestPhone = guestPhone
   292→        orderData.guestEmail = guestEmail || undefined
   293→        orderData.guestAddress = `${guestAddress}, ${guestCity}${guestDistrict ? `, ${guestDistrict}` : ''}`
   294→      }
   295→
   296→      const res = await fetch('/api/orders', {
   297→        method: 'POST',
   298→        headers: {
   299→          'Content-Type': 'application/json',
   300→          ...(useAppStore.getState().accessToken ? { 'Authorization': `Bearer ${useAppStore.getState().accessToken}` } : {})
   301→        },
   302→        body: JSON.stringify(orderData),
   303→      })
   304→
   305→      if (!res.ok) {
   306→        const data = await res.json()
   307→        throw new Error(data.error || 'Failed to place order')
   308→      }
   309→
   310→      const order = await res.json()
   311→      const orderId = order.data?.id || order.id
   312→      const orderNum = order.data?.orderNumber || order.orderNumber
   313→
   314→      if (paymentMethod === 'online') {
   315→        const initRes = await fetch('/api/payments/sslcommerz/initiate', {
   316→          method: 'POST',
   317→          headers: {
   318→            'Content-Type': 'application/json',
   319→            ...(useAppStore.getState().accessToken ? { 'Authorization': `Bearer ${useAppStore.getState().accessToken}` } : {})
   320→          },
   321→          body: JSON.stringify({ orderId }),
   322→        })
   323→        const initData = await initRes.json()

   325→        if (initData.success && initData.data?.gatewayPageURL) {
   326→          window.location.href = initData.data.gatewayPageURL
   327→          return
   328→        } else {
   329→          throw new Error(initData.error || 'Failed to initialize payment gateway')
   330→        }
   331→      } else {
   332→        await clearCart()
   333→        toast.success('Order placed successfully!')
   334→        navigate('order-confirmation', { orderId, orderNumber: orderNum })
   335→      }
   336→    } catch (error) {
   337→      toast.error(error instanceof Error ? error.message : 'Failed to place order')
   338→      setPlacing(false)
   339→    }
   340→  }
   341→
   342→  // ─── Empty Cart ───────────────────────────────────────────────────────────
   343→
   344→  if (cartItems.length === 0) {
   345→    return (
   346→      <div className="min-h-screen flex flex-col">
   347→        <div className="flex-1 flex items-center justify-center py-16 px-4">
   348→          <div className="text-center">
   349→            <div className="h-24 w-24 rounded-full bg-primary/10 flex items-center justify-center mx-auto mb-6">
   350→              <ShoppingBag className="h-12 w-12 text-primary/40" />
   351→            </div>
   352→            <h2 className="text-2xl font-bold mb-2">Your cart is empty</h2>
   353→            <p className="text-muted-foreground mb-6">Add products to your cart before checkout</p>
   354→            <Button onClick={() => navigate('products')} className="rounded-xl px-8 h-12">

   376→    </CardHeader>

   378→

   409→              ৳{(itemPrice * item.quantity).toLocaleString()}

   411→          </div>

   449→        >
   450→          ৳{Math.max(0, total).toLocaleString()}

   458→          </span>

   461→    </div>

   463→
   464→  const renderCouponSection = () => (

   470→      >
   471→        <Label className="text-xs font-semibold text-muted-foreground flex items-center gap-1.5 cursor-pointer">
   472→          <Tag className="h-3.5 w-3.5 text-primary" />
   473→          Coupon Code
   474→        </Label>
   475→        {showCouponInput ? (
   476→          <ChevronDown className="h-4 w-4 text-muted-foreground" />
   477→        ) : (
   478→          <ChevronRight className="h-4 w-4 text-muted-foreground" />
   479→        )}
   480→      </button>
   481→      <AnimatePresence>
   482→        {showCouponInput && (
   483→          <motion.div
   484→            initial={{ opacity: 0, height: 0 }}
   485→            animate={{ opacity: 1, height: 'auto' }}
   486→            exit={{ opacity: 0, height: 0 }}
   487→            className="overflow-hidden"
   488→          >
   489→            {couponApplied ? (
   490→              <div className="flex items-center justify-between p-2.5 rounded-lg bg-emerald-50 border border-emerald-200">
   491→                <div className="flex items-center gap-2">
   492→                  <Badge className="bg-emerald-100 text-emerald-800 border-emerald-200 hover:bg-emerald-100">
   493→                    <Gift className="h-3 w-3 mr-1" />
   494→                    {couponCode}
   495→                  </Badge>
   496→                  <span className="text-xs text-emerald-700 font-semibold">-৳{discount.toLocaleString()}</span>
   497→                </div>

   500→                  size="sm"
   501→                  onClick={() => { setCouponCode(''); setDiscount(0); setCouponApplied(false) }}
