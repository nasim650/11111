    90→function AnimatedSection({ children, className, delay = 0, style }: {

    95→}) {

   100→    <motion.div
   101→      ref={ref}
   102→      initial={{ opacity: 0, y: 40 }}
   103→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   104→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}

   135→// ── Flash Sale Countdown ──
   136→function FlashSaleCountdown() {
   137→  const [timeLeft, setTimeLeft] = useState({ hours: 0, minutes: 0, seconds: 0 })
   138→
   139→  useEffect(() => {
   140→    const getTargetTime = () => {
   141→      const now = new Date()
   142→      const end = new Date(now)
   143→      end.setHours(23, 59, 59, 999)
   144→      return end.getTime()
   145→    }
   146→
   147→    const updateCountdown = () => {
   148→      const target = getTargetTime()
   149→      const now = Date.now()
   150→      const diff = Math.max(target - now, 0)
   151→
   152→      setTimeLeft({
   153→        hours: Math.floor(diff / (1000 * 60 * 60)),
   154→        minutes: Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)),
   155→        seconds: Math.floor((diff % (1000 * 60)) / 1000),
   156→      })
   157→    }
   158→
   159→    updateCountdown()
   160→    const timer = setInterval(updateCountdown, 1000)
   161→    return () => clearInterval(timer)
   162→  }, [])
   163→
   164→  const pad = (n: number) => n.toString().padStart(2, '0')
   165→  const isEndingSoon = timeLeft.hours === 0 && timeLeft.minutes < 60
   166→
   167→  return (
   168→    <div className="flex items-center gap-2">
   169→      {isEndingSoon && (
   170→        <span
   171→          className="text-[10px] font-medium bg-white/15 text-white px-2 py-0.5 rounded-full mr-1"
   172→        >
   173→          Ending soon
   174→        </span>
   175→      )}
   176→      <div className="flex items-center gap-1">
   177→        <FlipDigit value={pad(timeLeft.hours)} label="ঘণ্টা" />
   178→        <span className="text-white font-bold text-xl mx-0.5 animate-pulse">:</span>
   179→        <FlipDigit value={pad(timeLeft.minutes)} label="মিনিট" />
   180→        <span className="text-white font-bold text-xl mx-0.5 animate-pulse">:</span>
   181→        <FlipDigit value={pad(timeLeft.seconds)} label="সেকেন্ড" />
   182→      </div>
   183→    </div>
   184→  )
   185→}

   261→// ── Hero Banner Carousel ──
   262→function HeroBannerCarousel({ banners, navigate }: {
   263→  banners: Banner[]
   264→  navigate: (page: PageName, params?: Record<string, unknown>) => void
   265→}) {
   266→  const [currentBanner, setCurrentBanner] = useState(0)
   267→  const [isPaused, setIsPaused] = useState(false)
   268→  const [progress, setProgress] = useState(0)
   269→  const [scrollY, setScrollY] = useState(0)
   270→  const intervalRef = useRef<NodeJS.Timeout | null>(null)
   271→
   272→  // Parallax scroll tracking
   273→  useEffect(() => {
   274→    const handleScroll = () => {
   275→      setScrollY(window.scrollY)
   276→    }
   277→    window.addEventListener('scroll', handleScroll, { passive: true })
   278→    return () => window.removeEventListener('scroll', handleScroll)
   279→  }, [])
   280→
   281→  // Auto-play with progress
   282→  useEffect(() => {
   283→    if (banners.length <= 1 || isPaused) return
   284→
   285→    const SLIDE_DURATION = 5000
   286→    const TICK = 50
   287→    let elapsed = 0
   288→
   289→    intervalRef.current = setInterval(() => {
   290→      elapsed += TICK
   291→      setProgress((elapsed / SLIDE_DURATION) * 100)
   292→
   293→      if (elapsed >= SLIDE_DURATION) {
   294→        elapsed = 0
   295→        setCurrentBanner((prev) => (prev + 1) % banners.length)
   296→        setProgress(0)
   297→      }
   298→    }, TICK)
   299→
   300→    return () => {
   301→      if (intervalRef.current) clearInterval(intervalRef.current)
   302→    }
   303→  }, [banners.length, isPaused, currentBanner])
   304→
   305→  const handlePrev = () => {
   306→    setCurrentBanner((prev) => (prev - 1 + banners.length) % banners.length)
   307→    setProgress(0)
   308→  }
   309→
   310→  const handleNext = () => {
   311→    setCurrentBanner((prev) => (prev + 1) % banners.length)
   312→    setProgress(0)
   313→  }
   314→
   315→  const handleDotClick = (idx: number) => {
   316→    setCurrentBanner(idx)
   317→    setProgress(0)
   318→  }
   319→
   320→  if (banners.length === 0) return null
   321→
   322→  // Parallax offset - capped at 60px
   323→  const parallaxOffset = Math.min(scrollY * 0.3, 60)
   324→
   325→  return (
   326→    <div
   327→      className="relative h-64 sm:h-80 md:h-[440px] lg:h-[520px] overflow-hidden bg-muted"
   328→      onMouseEnter={() => setIsPaused(true)}
   329→      onMouseLeave={() => setIsPaused(false)}
   330→    >
   331→      <AnimatePresence mode="wait">
   332→        <motion.div
   333→          key={currentBanner}
   334→          initial={{ opacity: 0, x: 60 }}
   335→          animate={{ opacity: 1, x: 0 }}
   336→          exit={{ opacity: 0, x: -60 }}
   337→          transition={{ duration: 0.5, ease: 'easeInOut' }}
   338→          className="absolute inset-0"
   339→        >
   340→          {banners[currentBanner].desktopImage ? (
   341→            <>
   342→              {/* Parallax image container */}
   343→              <div
   344→                className="absolute inset-0 w-full h-full"
   345→                style={{ transform: `translateY(${parallaxOffset}px) scale(1.1)` }}
   346→              >
   347→                <Image
   348→                  src={banners[currentBanner].desktopImage}
   349→                  alt={banners[currentBanner].title}
   350→                  fill
   351→                  className="object-cover"
   352→                  priority={currentBanner === 0}
   353→                  sizes="100vw"
   354→                />
   355→              </div>
   356→              {/* Gradient overlays - enhanced for text readability */}
   357→              <div className="absolute inset-0 bg-gradient-to-r from-black/80 via-black/50 to-black/20" />
   358→              <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-black/30" />

   510→// ── Category Section ──
   511→function CategorySection({ categories, navigate }: {
   512→  categories: Category[]
   513→  navigate: (page: PageName, params?: Record<string, unknown>) => void
   514→}) {
   515→  const scrollRef = useRef<HTMLDivElement>(null)
   516→  const [canScrollLeft, setCanScrollLeft] = useState(false)
   517→  const [canScrollRight, setCanScrollRight] = useState(true)
   518→
   519→  const checkScroll = useCallback(() => {
   520→    if (!scrollRef.current) return
   521→    const { scrollLeft, scrollWidth, clientWidth } = scrollRef.current
   522→    setCanScrollLeft(scrollLeft > 5)
   523→    setCanScrollRight(scrollLeft + clientWidth < scrollWidth - 5)
   524→  }, [])
   525→
   526→  useEffect(() => {
   527→    checkScroll()
   528→    const el = scrollRef.current
   529→    if (el) el.addEventListener('scroll', checkScroll, { passive: true })
   530→    return () => { if (el) el.removeEventListener('scroll', checkScroll) }
   531→  }, [checkScroll])
   532→
   533→  const handleScroll = (direction: 'left' | 'right') => {
   534→    if (!scrollRef.current) return
   535→    scrollRef.current.scrollBy({
   536→      left: direction === 'left' ? -300 : 300,

  1121→          const data = await discRes.json()
  1122→          const prods = extractItems<Product>(data)
