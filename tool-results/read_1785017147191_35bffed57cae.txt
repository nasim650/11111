   174→// Animated counter hook
   175→function useAnimatedCounter(target: number, duration: number = 800) {
   176→  const [count, setCount] = useState(0)
   177→  const prevTarget = useRef(0)
   178→
   179→  useEffect(() => {
   180→    if (target === prevTarget.current) return
   181→    prevTarget.current = target
   182→
   183→    const start = 0
   184→    const diff = target - start
   185→    const startTime = performance.now()
   186→
   187→    function step(currentTime: number) {
   188→      const elapsed = currentTime - startTime
   189→      const progress = Math.min(elapsed / duration, 1)
   190→      const eased = 1 - Math.pow(1 - progress, 3)
   191→      setCount(Math.round(start + diff * eased))
   192→      if (progress < 1) {
   193→        requestAnimationFrame(step)
   194→      }
   195→    }
   196→
   197→    requestAnimationFrame(step)
   198→  }, [target, duration])
   199→
   200→  return count
   201→}
