   276→  function validateConfig(): boolean {
   277→    const newErrors: FormErrors = {}
   278→    if (!config.siteName.trim()) newErrors.siteName = 'Site name is required'
   279→    if (!config.email.trim()) newErrors.email = 'Email is required'
   280→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(config.email)) newErrors.email = 'Invalid email format'
   281→    if (config.shippingCharge < 0) newErrors.shippingCharge = 'Shipping charge cannot be negative'
   282→    if (config.freeShippingThreshold < 0) newErrors.freeShippingThreshold = 'Free shipping threshold cannot be negative'
   283→    if (config.taxRate < 0 || config.taxRate > 100) newErrors.taxRate = 'Tax rate must be between 0-100'
   284→    if ((config.sessionTimeout ?? 0) < 5) newErrors.sessionTimeout = 'Session timeout must be at least 5 minutes'
   285→    setErrors(newErrors)
   286→    return Object.keys(newErrors).length === 0
   287→  }

   384→      return next
   385→    })
   386→  }

   392→  async function handleImageUpload(field: 'logo' | 'favicon') {
   393→    const input = document.createElement('input')
   394→    input.type = 'file'
   395→    input.accept = 'image/*'
   396→    input.onchange = async (e) => {
   397→      const file = (e.target as HTMLInputElement).files?.[0]
   398→      if (!file) return
   399→      const formData = new FormData()
   400→      formData.append('files', file)
   401→      try {
   402→        const res = await fetch('/api/uploads', { method: 'POST', body: formData })
   403→        if (!res.ok) throw new Error('Upload failed')
   404→        const data = await res.json()
   405→        const files = data.files ?? data ?? []
   406→        if (files.length > 0) {
   407→          updateConfig({ [field]: files[0].url })
   408→          toast.success(`${field === 'logo' ? 'Logo' : 'Favicon'} uploaded successfully`)
   409→        }
   410→      } catch {
   411→        toast.error('Image upload failed')
   412→      }
   413→    }
   414→    input.click()
   415→  }

   417→  function handlePasswordChange() {
   418→    const errs: FormErrors = {}
   419→    if (!currentPassword) errs.currentPassword = 'Current password is required'
   420→    if (!newPassword) errs.newPassword = 'New password is required'
   421→    else if (newPassword.length < 8) errs.newPassword = 'Password must be at least 8 characters'
   422→    if (newPassword !== confirmPassword) errs.confirmPassword = 'Passwords do not match'
   423→    setPasswordErrors(errs)
   424→    if (Object.keys(errs).length > 0) return
   425→
   426→    // Simulate password change
   427→    toast.success('Password changed successfully!')
   428→    setCurrentPassword('')
   429→    setNewPassword('')
   430→    setConfirmPassword('')
   431→  }
