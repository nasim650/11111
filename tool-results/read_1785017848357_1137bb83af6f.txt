   121→function AnimatedSection({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   122→  const ref = useRef<HTMLDivElement>(null)
   123→  const isInView = useInView(ref, { once: true, margin: '-50px' })
   124→  return (
   125→    <motion.div
   126→      ref={ref}
   127→      initial={{ opacity: 0, y: 30 }}
   128→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   129→      transition={{ duration: 0.5, delay, ease: 'easeOut' }}
   130→      className={className}
   131→    >
   132→      {children}
   133→    </motion.div>
   134→  )
   135→}

   137→// ─── Fullscreen Image Viewer ──────────────────────────────────────────────────
   138→
   139→function FullscreenViewer({
   140→  images,
   141→  initialIndex,
   142→  open,
   143→  onClose,
   144→}: {
   145→  images: Array<{ id: string; url: string; alt?: string }>
   146→  initialIndex: number
   147→  open: boolean
   148→  onClose: () => void
   149→}) {
   150→  const [idx, setIdx] = useState(initialIndex)
   151→
   152→  // Sync idx when the dialog opens or initialIndex changes (lint fix preserved)
   153→  if (open && idx !== initialIndex) {
   154→    setIdx(initialIndex)
   155→  }
   156→
   157→  useEffect(() => {
   158→    if (!open) return
   159→    const handler = (e: KeyboardEvent) => {
   160→      if (e.key === 'ArrowRight') setIdx((i) => Math.min(images.length - 1, i + 1))
   161→      if (e.key === 'ArrowLeft') setIdx((i) => Math.max(0, i - 1))
   162→      if (e.key === 'Escape') onClose()
   163→    }
   164→    window.addEventListener('keydown', handler)
   165→    return () => window.removeEventListener('keydown', handler)
   166→  }, [open, images.length, onClose])
   167→
   168→  return (
   169→    <Dialog open={open} onOpenChange={(v) => { if (!v) onClose() }}>
   170→      <DialogContent className="max-w-5xl w-full p-0 bg-black/95 border-none overflow-hidden">
   171→        <DialogTitle className="sr-only">Image Viewer</DialogTitle>
   172→        <div className="relative aspect-square w-full">
   173→          <AnimatePresence mode="wait">
   174→            <motion.div
   175→              key={idx}
   176→              initial={{ opacity: 0, scale: 0.95 }}
   177→              animate={{ opacity: 1, scale: 1 }}
   178→              exit={{ opacity: 0, scale: 0.95 }}
   179→              transition={{ duration: 0.3 }}
   180→              className="absolute inset-0"
   181→            >
   182→              <Image
   183→                src={images[idx]?.url || '/placeholder.png'}
   184→                alt={images[idx]?.alt || 'Product image'}
   185→                fill
   186→                className="object-contain"
   187→                sizes="90vw"
   188→                priority
   189→              />
   190→            </motion.div>
   191→          </AnimatePresence>
   192→          {/* Close */}
   193→          <button
   194→            onClick={onClose}
   195→            className="absolute top-4 right-4 z-10 h-10 w-10 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors"
   196→            aria-label="Close viewer"
   197→          >
   198→            <X className="h-5 w-5" />
   199→          </button>
   200→          {/* Nav arrows */}
   201→          {images.length > 1 && (
   202→            <>
   203→              <button
   204→                onClick={() => setIdx((i) => Math.max(0, i - 1))}
   205→                disabled={idx === 0}
   206→                className="absolute left-4 top-1/2 -translate-y-1/2 z-10 h-12 w-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors disabled:opacity-30"
   207→                aria-label="Previous image"
   208→              >
   209→                <ChevronLeft className="h-6 w-6" />
   210→              </button>
   211→              <button
   212→                onClick={() => setIdx((i) => Math.min(images.length - 1, i + 1))}
   213→                disabled={idx === images.length - 1}
   214→                className="absolute right-4 top-1/2 -translate-y-1/2 z-10 h-12 w-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors disabled:opacity-30"
   215→                aria-label="Next image"
   216→              >
   217→                <ChevronRight className="h-6 w-6" />
   218→              </button>
   219→            </>
   220→          )}
   221→          {/* Counter */}
   222→          <div className="absolute bottom-4 left-1/2 -translate-x-1/2 z-10 bg-black/60 backdrop-blur-sm text-white text-sm px-3 py-1.5 rounded-full">
   223→            {idx + 1} / {images.length}
   224→          </div>
   225→          {/* Thumbnails */}
   226→          {images.length > 1 && (
   227→            <div className="absolute bottom-14 left-1/2 -translate-x-1/2 z-10 flex gap-2 max-w-md overflow-x-auto px-2 py-1 scrollbar-hide">
   228→              {images.map((img, i) => (
   229→                <button
   230→                  key={img.id}
   231→                  onClick={() => setIdx(i)}
   232→                  className={`relative h-12 w-12 rounded-md overflow-hidden flex-shrink-0 border-2 transition-all ${
   233→                    i === idx ? 'border-white ring-2 ring-white/30' : 'border-white/30 opacity-60 hover:opacity-90'
   234→                  }`}
   235→                >
   236→                  <Image src={img.url} alt={img.alt || ''} fill className="object-cover" sizes="48px" />
   237→                </button>
   238→              ))}
   239→            </div>
   240→          )}
   241→        </div>
   242→      </DialogContent>
   243→    </Dialog>
   244→  )
   245→}

   247→// ─── Scroll-to-Top Button ────────────────────────────────────────────────────

   249→function ScrollToTopButton() {
   250→  const [visible, setVisible] = useState(false)
   251→
   252→  useEffect(() => {
   253→    const handleScroll = () => setVisible(window.scrollY > 600)
   254→    window.addEventListener('scroll', handleScroll, { passive: true })
   255→    return () => window.removeEventListener('scroll', handleScroll)
   256→  }, [])
   257→
   258→  return (
   259→    <AnimatePresence>
   260→      {visible && (
   261→        <motion.button
   262→          initial={{ opacity: 0, scale: 0.8 }}
   263→          animate={{ opacity: 1, scale: 1 }}
   264→          exit={{ opacity: 0, scale: 0.8 }}
   265→          onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   266→          className="fixed bottom-20 md:bottom-8 right-4 z-40 h-10 w-10 rounded-full bg-emerald-600 text-white shadow-lg shadow-emerald-600/30 flex items-center justify-center hover:bg-emerald-700 transition-colors"
   267→          aria-label="Scroll to top"
   268→        >
   269→          <ArrowUp className="h-4 w-4" />
   270→        </motion.button>
   271→      )}
   272→    </AnimatePresence>
   273→  )
   274→}
