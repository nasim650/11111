   188→   188→  function generateSlug(name: string): string {
   189→   189→    return name
   190→   190→      .toLowerCase()
   191→   191→      .replace(/[^a-z0-9\s-]/g, '')
   192→   192→      .replace(/\s+/g, '-')

   220→   220→  async function handleSave() {
   221→   221→    if (!editingProduct?.name) {
   222→   222→      toast.error('Product name is required')
   223→   223→      return
   224→   224→    }
   225→   225→    setSaving(true)
   226→   226→    try {
   227→   227→      const payload = {
   228→   228→        ...editingProduct,
   229→   229→        slug: editingProduct.slug || generateSlug(editingProduct.name),
   230→   230→        price: Number(editingProduct.price) || 0,
   231→   231→        discountPrice: editingProduct.discountPrice ? Number(editingProduct.discountPrice) : null,
   232→   232→        stock: Number(editingProduct.stock) || 0,
   233→   233→      }
   234→   234→
   235→   235→      const isEdit = !!editingProduct.id
   236→   236→      const url = isEdit ? `/api/products/${editingProduct.id}` : '/api/products'
   237→   237→      const method = isEdit ? 'PUT' : 'POST'
   238→   238→
   239→   239→      const res = await fetch(url, {
   240→   240→        method,
   241→   241→        headers: { 'Content-Type': 'application/json' },
   242→   242→        body: JSON.stringify(payload),
   243→   243→      })
   244→   244→
   245→   245→      if (!res.ok) {
   246→   246→        const data = await res.json()
   247→   247→        throw new Error(data.error || 'Failed to save product')
   248→   248→      }
   249→   249→
   250→   250→      toast.success(isEdit ? 'Product updated' : 'Product created')
   251→   251→      setDialogOpen(false)
   252→   252→      fetchProducts()
   253→   253→    } catch (err) {
   254→   254→      toast.error(err instanceof Error ? err.message : 'Failed to save product')
   255→   255→    } finally {
   256→   256→      setSaving(false)
   257→   257→    }
   258→   258→  }
