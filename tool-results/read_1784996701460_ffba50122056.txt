     1→'use client'
     2→
     3→import React, { useEffect, useState, useCallback } from 'react'
     4→import { Card, CardContent } from '@/components/ui/card'
     5→import { Button } from '@/components/ui/button'
     6→import { Badge } from '@/components/ui/badge'
     7→import { Input } from '@/components/ui/input'
     8→import {
     9→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    10→} from '@/components/ui/table'
    11→import {
    12→  Dialog, DialogContent, DialogHeader, DialogTitle,
    13→} from '@/components/ui/dialog'
    14→import {
    15→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    16→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    17→} from '@/components/ui/alert-dialog'
    18→import {
    19→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    20→} from '@/components/ui/select'
    21→import { Skeleton } from '@/components/ui/skeleton'
    22→import { Separator } from '@/components/ui/separator'
    23→import { toast } from 'sonner'
    24→import { extractItems, extractPagination } from '@/lib/api'
    25→import { useAppStore } from '@/lib/store'
    26→import {
    27→  Mail, Eye, Trash2, Search, MailOpen, MailX,
    28→  Inbox, ChevronLeft, ChevronRight, Phone, User, Clock,
    29→} from 'lucide-react'
    30→
    31→interface ContactMessage {
    32→  id: string
    33→  name: string
    34→  email: string
    35→  phone?: string | null
    36→  subject?: string | null
    37→  message: string
    38→  isRead: boolean
    39→  createdAt: string
    40→  updatedAt: string
    41→}
    42→
    43→export default function AdminContactMessagesPage() {
    44→  const [messages, setMessages] = useState<ContactMessage[]>([])
    45→  const [loading, setLoading] = useState(true)
    46→  const [search, setSearch] = useState('')
    47→  const [readFilter, setReadFilter] = useState<string>('all')
    48→  const [page, setPage] = useState(1)
    49→  const [totalPages, setTotalPages] = useState(1)
    50→  const [total, setTotal] = useState(0)
    51→  const [unreadCount, setUnreadCount] = useState(0)
    52→  const [selectedMessage, setSelectedMessage] = useState<ContactMessage | null>(null)
    53→  const [detailOpen, setDetailOpen] = useState(false)
    54→  const [deleteId, setDeleteId] = useState<string | null>(null)
    55→  const [markingId, setMarkingId] = useState<string | null>(null)
    56→
    57→  const token = useAppStore((state) => state.accessToken)
    58→
    59→  const getHeaders = useCallback(() => {
    60→    const headers: Record<string, string> = { 'Content-Type': 'application/json' }
    61→    if (token) headers['Authorization'] = `Bearer ${token}`
    62→    return headers
    63→  }, [token])
    64→
    65→  const fetchMessages = useCallback(async () => {
    66→    try {
    67→      setLoading(true)
    68→      const params = new URLSearchParams()
    69→      params.set('page', String(page))
    70→      params.set('limit', '10')
    71→      if (readFilter === 'unread') params.set('isRead', 'false')
    72→      else if (readFilter === 'read') params.set('isRead', 'true')
    73→
    74→      const res = await fetch(`/api/contact?${params.toString()}`, {
    75→        headers: getHeaders(),
    76→      })
    77→      if (res.ok) {
    78→        const data = await res.json()
    79→        setMessages(extractItems<ContactMessage>(data))
    80→        const pagination = extractPagination(data)
    81→        setTotalPages(pagination.totalPages || 1)
    82→        setTotal(pagination.total)
    83→      }
    84→    } catch {
    85→      toast.error('Failed to fetch messages')
    86→    } finally {
    87→      setLoading(false)
    88→    }
    89→  }, [page, readFilter, getHeaders])
    90→
    91→  const fetchUnreadCount = useCallback(async () => {
    92→    try {
    93→      const res = await fetch('/api/contact?limit=1&isRead=false', {
    94→        headers: getHeaders(),
    95→      })
    96→      if (res.ok) {
    97→        const data = await res.json()
    98→        const pagination = extractPagination(data)
    99→        setUnreadCount(pagination.total)
   100→      }
   101→    } catch {
   102→      // silent
   103→    }
   104→  }, [getHeaders])
   105→
   106→  useEffect(() => {
   107→    fetchMessages()
   108→  }, [fetchMessages])
   109→
   110→  useEffect(() => {
   111→    fetchUnreadCount()
   112→  }, [fetchUnreadCount])
   113→
   114→  // Refetch unread count when messages change
   115→  useEffect(() => {
   116→    fetchUnreadCount()
   117→  }, [messages, fetchUnreadCount])
   118→
   119→  async function toggleReadStatus(msg: ContactMessage) {
   120→    setMarkingId(msg.id)
   121→    try {
   122→      const res = await fetch(`/api/contact/${msg.id}`, {
   123→        method: 'PATCH',
   124→        headers: getHeaders(),
   125→        body: JSON.stringify({ isRead: !msg.isRead }),
   126→      })
   127→      if (!res.ok) throw new Error('Failed to update')
   128→      toast.success(msg.isRead ? 'Marked as unread' : 'Marked as read')
   129→      fetchMessages()
   130→    } catch {
   131→      toast.error('Failed to update message status')
   132→    } finally {
   133→      setMarkingId(null)
   134→    }
   135→  }
   136→
   137→  async function handleDelete() {
   138→    if (!deleteId) return
   139→    try {
   140→      const res = await fetch(`/api/contact/${deleteId}`, {
   141→        method: 'DELETE',
   142→        headers: getHeaders(),
   143→      })
   144→      if (!res.ok) throw new Error('Failed to delete')
   145→      toast.success('Message deleted')
   146→      fetchMessages()
   147→    } catch {
   148→      toast.error('Failed to delete message')
   149→    } finally {
   150→      setDeleteId(null)
   151→    }
   152→  }
   153→
   154→  function openDetail(msg: ContactMessage) {
   155→    setSelectedMessage(msg)
   156→    setDetailOpen(true)
   157→    // Auto-mark as read when opened
   158→    if (!msg.isRead) {
   159→      fetch(`/api/contact/${msg.id}`, {
   160→        method: 'PATCH',
   161→        headers: getHeaders(),
   162→        body: JSON.stringify({ isRead: true }),
   163→      }).then(() => {
   164→        fetchMessages()
   165→      }).catch(() => {
   166→        // silent
   167→      })
   168→    }
   169→  }
   170→
   171→  // Client-side search filtering
   172→  const filteredMessages = messages.filter((msg) => {
   173→    if (!search) return true
   174→    const q = search.toLowerCase()
   175→    return (
   176→      msg.name.toLowerCase().includes(q) ||
   177→      msg.email.toLowerCase().includes(q) ||
   178→      (msg.subject && msg.subject.toLowerCase().includes(q))
   179→    )
   180→  })
   181→
   182→  const readCount = total - unreadCount
   183→
   184→  function formatDate(dateStr: string) {
   185→    const date = new Date(dateStr)
   186→    const now = new Date()
   187→    const diff = now.getTime() - date.getTime()
   188→    const days = Math.floor(diff / (1000 * 60 * 60 * 24))
   189→
   190→    if (days === 0) {
   191→      const hours = Math.floor(diff / (1000 * 60 * 60))
   192→      if (hours === 0) {
   193→        const minutes = Math.floor(diff / (1000 * 60))
   194→        return minutes <= 1 ? 'Just now' : `${minutes}m ago`
   195→      }
   196→      return `${hours}h ago`
   197→    }
   198→    if (days === 1) return 'Yesterday'
   199→    if (days < 7) return `${days}d ago`
   200→    return date.toLocaleDateString('en-US', {
   201→      month: 'short',
   202→      day: 'numeric',
   203→      year: date.getFullYear() !== now.getFullYear() ? 'numeric' : undefined,
   204→    })
   205→  }
   206→
   207→  return (
   208→    <div className="space-y-4">
   209→      {/* Stats Row */}
   210→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   211→        <Card className="border-l-4 border-l-green-600">
   212→          <CardContent className="p-4 flex items-center gap-3">
   213→            <div className="flex h-10 w-10 items-center justify-center rounded-full bg-green-100 text-green-700">
   214→              <Inbox className="size-5" />
   215→            </div>
   216→            <div>
   217→              <p className="text-2xl font-bold">{total}</p>
   218→              <p className="text-xs text-muted-foreground">Total Messages</p>
   219→            </div>
   220→          </CardContent>
   221→        </Card>
   222→        <Card className="border-l-4 border-l-amber-500">
   223→          <CardContent className="p-4 flex items-center gap-3">
   224→            <div className="flex h-10 w-10 items-center justify-center rounded-full bg-amber-100 text-amber-700">
   225→              <Mail className="size-5" />
   226→            </div>
   227→            <div>
   228→              <p className="text-2xl font-bold">{unreadCount}</p>
   229→              <p className="text-xs text-muted-foreground">Unread</p>
   230→            </div>
   231→          </CardContent>
   232→        </Card>
   233→        <Card className="border-l-4 border-l-emerald-500">
   234→          <CardContent className="p-4 flex items-center gap-3">
   235→            <div className="flex h-10 w-10 items-center justify-center rounded-full bg-emerald-100 text-emerald-700">
   236→              <MailOpen className="size-5" />
   237→            </div>
   238→            <div>
   239→              <p className="text-2xl font-bold">{readCount}</p>
   240→              <p className="text-xs text-muted-foreground">Read</p>
   241→            </div>
   242→          </CardContent>
   243→        </Card>
   244→      </div>
   245→
   246→      {/* Filters */}
   247→      <div className="flex flex-col sm:flex-row gap-3">
   248→        <div className="relative flex-1">
   249→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   250→          <Input
   251→            placeholder="Search by name, email, or subject..."
   252→            value={search}
   253→            onChange={(e) => setSearch(e.target.value)}
   254→            className="pl-9"
   255→          />
   256→        </div>
   257→        <Select value={readFilter} onValueChange={(v) => { setReadFilter(v); setPage(1) }}>
   258→          <SelectTrigger className="w-full sm:w-40">
   259→            <SelectValue placeholder="Filter status" />
   260→          </SelectTrigger>
   261→          <SelectContent>
   262→            <SelectItem value="all">All Messages</SelectItem>
   263→            <SelectItem value="unread">
   264→              <span className="flex items-center gap-2">
   265→                <span className="size-2 rounded-full bg-amber-500" />
   266→                Unread
   267→              </span>
   268→            </SelectItem>
   269→            <SelectItem value="read">
   270→              <span className="flex items-center gap-2">
   271→                <span className="size-2 rounded-full bg-green-500" />
   272→                Read
   273→              </span>
   274→            </SelectItem>
   275→          </SelectContent>
   276→        </Select>
   277→      </div>
   278→
   279→      {/* Messages Table / Cards */}
   280→      <Card>
   281→        <CardContent className="p-0">
   282→          {loading ? (
   283→            <div className="p-6 space-y-3">
   284→              {[...Array(5)].map((_, i) => (
   285→                <Skeleton key={i} className="h-14 w-full" />
   286→              ))}
   287→            </div>
   288→          ) : filteredMessages.length === 0 ? (
   289→            <div className="text-center py-16">
   290→              <Inbox className="mx-auto size-12 text-muted-foreground/40 mb-3" />
   291→              <p className="text-muted-foreground font-medium">No messages found</p>
   292→              <p className="text-sm text-muted-foreground/60 mt-1">
   293→                {search ? 'Try adjusting your search terms' : 'Contact messages will appear here'}
   294→              </p>
   295→            </div>
   296→          ) : (
   297→            <>
   298→              {/* Desktop Table */}
   299→              <div className="hidden md:block">
   300→                <Table>
   301→                  <TableHeader>
   302→                    <TableRow>
   303→                      <TableHead className="w-8" />
   304→                      <TableHead>Sender</TableHead>
   305→                      <TableHead>Subject</TableHead>
   306→                      <TableHead>Date</TableHead>
   307→                      <TableHead>Status</TableHead>
   308→                      <TableHead className="w-28">Actions</TableHead>
   309→                    </TableRow>
   310→                  </TableHeader>
   311→                  <TableBody>
   312→                    {filteredMessages.map((msg) => (
   313→                      <TableRow
   314→                        key={msg.id}
   315→                        className={`cursor-pointer hover:bg-muted/50 ${!msg.isRead ? 'bg-amber-50/50 dark:bg-amber-950/10' : ''}`}
   316→                        onClick={() => openDetail(msg)}
   317→                      >
   318→                        <TableCell>
   319→                          {!msg.isRead && (
   320→                            <span className="flex size-2.5 rounded-full bg-amber-500" />
   321→                          )}
   322→                        </TableCell>
   323→                        <TableCell>
   324→                          <div className={!msg.isRead ? 'font-semibold' : ''}>
   325→                            <p className={`text-sm ${!msg.isRead ? 'font-semibold' : ''}`}>
   326→                              {msg.name}
   327→                            </p>
   328→                            <p className="text-xs text-muted-foreground">{msg.email}</p>
   329→                          </div>
   330→                        </TableCell>
   331→                        <TableCell>
   332→                          <p className={`text-sm max-w-xs truncate ${!msg.isRead ? 'font-semibold' : ''}`}>
   333→                            {msg.subject || 'No subject'}
   334→                          </p>
   335→                          <p className="text-xs text-muted-foreground max-w-xs truncate">
   336→                            {msg.message.substring(0, 60)}{msg.message.length > 60 ? '...' : ''}
   337→                          </p>
   338→                        </TableCell>
   339→                        <TableCell className="text-xs text-muted-foreground whitespace-nowrap">
   340→                          {formatDate(msg.createdAt)}
   341→                        </TableCell>
   342→                        <TableCell>
   343→                          <Badge
   344→                            variant="secondary"
   345→                            className={
   346→                              !msg.isRead
   347→                                ? 'bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-400'
   348→                                : 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400'
   349→                            }
   350→                          >
   351→                            {!msg.isRead ? 'Unread' : 'Read'}
   352→                          </Badge>
   353→                        </TableCell>
   354→                        <TableCell>
   355→                          <div className="flex gap-1" onClick={(e) => e.stopPropagation()}>
   356→                            <Button
   357→                              variant="ghost"
   358→                              size="icon"
   359→                              className="size-7"
   360→                              onClick={() => openDetail(msg)}
   361→                              title="View details"
   362→                            >
   363→                              <Eye className="size-3" />
   364→                            </Button>
   365→                            <Button
   366→                              variant="ghost"
   367→                              size="icon"
   368→                              className="size-7"
   369→                              disabled={markingId === msg.id}
   370→                              onClick={() => toggleReadStatus(msg)}
   371→                              title={msg.isRead ? 'Mark as unread' : 'Mark as read'}
   372→                            >
   373→                              {msg.isRead ? (
   374→                                <MailX className="size-3 text-amber-600" />
   375→                              ) : (
   376→                                <MailOpen className="size-3 text-green-600" />
   377→                              )}
   378→                            </Button>
   379→                            <Button
   380→                              variant="ghost"
   381→                              size="icon"
   382→                              className="size-7 text-destructive"
   383→                              onClick={() => setDeleteId(msg.id)}
   384→                              title="Delete"
   385→                            >
   386→                              <Trash2 className="size-3" />
   387→                            </Button>
   388→                          </div>
   389→                        </TableCell>
   390→                      </TableRow>
   391→                    ))}
   392→                  </TableBody>
   393→                </Table>
   394→              </div>
   395→
   396→              {/* Mobile Cards */}
   397→              <div className="md:hidden divide-y">
   398→                {filteredMessages.map((msg) => (
   399→                  <div
   400→                    key={msg.id}
   401→                    className={`p-4 cursor-pointer active:bg-muted/50 ${!msg.isRead ? 'bg-amber-50/50 dark:bg-amber-950/10' : ''}`}
   402→                    onClick={() => openDetail(msg)}
   403→                  >
   404→                    <div className="flex items-start gap-3">
   405→                      {/* Unread indicator */}
   406→                      <div className="pt-1.5">
   407→                        {!msg.isRead ? (
   408→                          <span className="flex size-2.5 rounded-full bg-amber-500" />
   409→                        ) : (
   410→                          <span className="flex size-2.5" />
   411→                        )}
   412→                      </div>
   413→
   414→                      <div className="flex-1 min-w-0">
   415→                        <div className="flex items-center justify-between gap-2">
   416→                          <p className={`text-sm truncate ${!msg.isRead ? 'font-semibold' : ''}`}>
   417→                            {msg.name}
   418→                          </p>
   419→                          <span className="text-xs text-muted-foreground whitespace-nowrap">
   420→                            {formatDate(msg.createdAt)}
   421→                          </span>
   422→                        </div>
   423→                        <p className="text-xs text-muted-foreground truncate">{msg.email}</p>
   424→                        <p className={`text-sm mt-1 truncate ${!msg.isRead ? 'font-semibold' : ''}`}>
   425→                          {msg.subject || 'No subject'}
   426→                        </p>
   427→                        <p className="text-xs text-muted-foreground mt-0.5 line-clamp-1">
   428→                          {msg.message.substring(0, 80)}{msg.message.length > 80 ? '...' : ''}
   429→                        </p>
   430→                        <div className="flex items-center justify-between mt-2">
   431→                          <Badge
   432→                            variant="secondary"
   433→                            className={
   434→                              !msg.isRead
   435→                                ? 'bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-400 text-[10px]'
   436→                                : 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400 text-[10px]'
   437→                            }
   438→                          >
   439→                            {!msg.isRead ? 'Unread' : 'Read'}
   440→                          </Badge>
   441→                          <div className="flex gap-1" onClick={(e) => e.stopPropagation()}>
   442→                            <Button
   443→                              variant="ghost"
   444→                              size="icon"
   445→                              className="size-7"
   446→                              disabled={markingId === msg.id}
   447→                              onClick={() => toggleReadStatus(msg)}
   448→                              title={msg.isRead ? 'Mark as unread' : 'Mark as read'}
   449→                            >
   450→                              {msg.isRead ? (
   451→                                <MailX className="size-3 text-amber-600" />
   452→                              ) : (
   453→                                <MailOpen className="size-3 text-green-600" />
   454→                              )}
   455→                            </Button>
   456→                            <Button
   457→                              variant="ghost"
   458→                              size="icon"
   459→                              className="size-7 text-destructive"
   460→                              onClick={() => setDeleteId(msg.id)}
   461→                              title="Delete"
   462→                            >
   463→                              <Trash2 className="size-3" />
   464→                            </Button>
   465→                          </div>
   466→                        </div>
   467→                      </div>
   468→                    </div>
   469→                  </div>
   470→                ))}
   471→              </div>
   472→            </>
   473→          )}
   474→        </CardContent>
   475→      </Card>
   476→
   477→      {/* Pagination */}
   478→      {totalPages > 1 && (
   479→        <div className="flex items-center justify-between">
   480→          <p className="text-sm text-muted-foreground">
   481→            Page {page} of {totalPages} ({total} message{total !== 1 ? 's' : ''})
   482→          </p>
   483→          <div className="flex gap-2">
   484→            <Button
   485→              variant="outline"
   486→              size="sm"
   487→              disabled={page <= 1}
   488→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   489→            >
   490→              <ChevronLeft className="size-4 mr-1" /> Previous
   491→            </Button>
   492→            <Button
   493→              variant="outline"
   494→              size="sm"
   495→              disabled={page >= totalPages}
   496→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   497→            >
   498→              Next <ChevronRight className="size-4 ml-1" />
   499→            </Button>
   500→          </div>
   501→        </div>
   502→      )}
   503→
   504→      {/* Detail Dialog */}
   505→      <Dialog open={detailOpen} onOpenChange={setDetailOpen}>
   506→        <DialogContent className="max-w-lg">
   507→          <DialogHeader>
   508→            <DialogTitle className="flex items-center gap-2">
   509→              <Mail className="size-5 text-primary" />
   510→              Message Details
   511→            </DialogTitle>
   512→          </DialogHeader>
   513→          {selectedMessage && (
   514→            <div className="space-y-4">
   515→              {/* Status Badge */}
   516→              <div className="flex items-center gap-2">
   517→                <Badge
   518→                  variant="secondary"
   519→                  className={
   520→                    !selectedMessage.isRead
   521→                      ? 'bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-400'
   522→                      : 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400'
   523→                  }
   524→                >
   525→                  {!selectedMessage.isRead ? 'Unread' : 'Read'}
   526→                </Badge>
   527→                <span className="text-xs text-muted-foreground flex items-center gap-1">
   528→                  <Clock className="size-3" />
   529→                  {new Date(selectedMessage.createdAt).toLocaleString()}
   530→                </span>
   531→              </div>
   532→
   533→              <Separator />
   534→
   535→              {/* Sender Info */}
   536→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   537→                <div className="flex items-start gap-2">
   538→                  <User className="size-4 text-muted-foreground mt-0.5" />
   539→                  <div>
   540→                    <p className="text-xs text-muted-foreground">Name</p>
   541→                    <p className="text-sm font-medium">{selectedMessage.name}</p>
   542→                  </div>
   543→                </div>
   544→                <div className="flex items-start gap-2">
   545→                  <Mail className="size-4 text-muted-foreground mt-0.5" />
   546→                  <div>
   547→                    <p className="text-xs text-muted-foreground">Email</p>
   548→                    <a
   549→                      href={`mailto:${selectedMessage.email}`}
   550→                      className="text-sm font-medium text-primary hover:underline"
   551→                    >
   552→                      {selectedMessage.email}
   553→                    </a>
   554→                  </div>
   555→                </div>
   556→                {selectedMessage.phone && (
   557→                  <div className="flex items-start gap-2">
   558→                    <Phone className="size-4 text-muted-foreground mt-0.5" />
   559→                    <div>
   560→                      <p className="text-xs text-muted-foreground">Phone</p>
   561→                      <a
   562→                        href={`tel:${selectedMessage.phone}`}
   563→                        className="text-sm font-medium text-primary hover:underline"
   564→                      >
   565→                        {selectedMessage.phone}
