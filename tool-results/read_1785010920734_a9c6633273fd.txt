   117→   117→function AnimatedSection({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   118→   118→  const ref = useRef<HTMLDivElement>(null)
   119→   119→  const isInView = useInView(ref, { once: true, margin: '-50px' })
   120→   120→  return (
   121→   121→    <motion.div
   122→   122→      ref={ref}
   123→   123→      initial={{ opacity: 0, y: 30 }}
   124→   124→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   125→   125→      transition={{ duration: 0.5, delay, ease: 'easeOut' }}
   126→   126→      className={className}
   127→   127→    >
   128→   128→      {children}
   129→   129→    </motion.div>
   130→   130→  )
   131→   131→}

   135→   135→function FullscreenViewer({
   136→   136→  images,
   137→   137→  initialIndex,
   138→   138→  open,
   139→   139→  onClose,
   140→   140→}: {
   141→   141→  images: Array<{ id: string; url: string; alt?: string }>
   142→   142→  initialIndex: number
   143→   143→  open: boolean
   144→   144→  onClose: () => void
   145→   145→}) {
   146→   146→  const [idx, setIdx] = useState(initialIndex)
   147→   147→
   148→   148→  // Sync idx when the dialog opens or initialIndex changes (lint fix preserved)
   149→   149→  if (open && idx !== initialIndex) {
   150→   150→    setIdx(initialIndex)
   151→   151→  }
   152→   152→
   153→   153→  useEffect(() => {
   154→   154→    if (!open) return
   155→   155→    const handler = (e: KeyboardEvent) => {
   156→   156→      if (e.key === 'ArrowRight') setIdx((i) => Math.min(images.length - 1, i + 1))
   157→   157→      if (e.key === 'ArrowLeft') setIdx((i) => Math.max(0, i - 1))
   158→   158→      if (e.key === 'Escape') onClose()
   159→   159→    }
   160→   160→    window.addEventListener('keydown', handler)
   161→   161→    return () => window.removeEventListener('keydown', handler)
   162→   162→  }, [open, images.length, onClose])
   163→   163→
   164→   164→  return (
   165→   165→    <Dialog open={open} onOpenChange={(v) => { if (!v) onClose() }}>
   166→   166→      <DialogContent className="max-w-5xl w-full p-0 bg-black/95 border-none overflow-hidden">
   167→   167→        <DialogTitle className="sr-only">Image Viewer</DialogTitle>
   168→   168→        <div className="relative aspect-square w-full">
   169→   169→          <AnimatePresence mode="wait">
   170→   170→            <motion.div
   171→   171→              key={idx}
   172→   172→              initial={{ opacity: 0, scale: 0.95 }}
   173→   173→              animate={{ opacity: 1, scale: 1 }}
   174→   174→              exit={{ opacity: 0, scale: 0.95 }}
   175→   175→              transition={{ duration: 0.3 }}
   176→   176→              className="absolute inset-0"
   177→   177→            >
   178→   178→              <Image
   179→   179→                src={images[idx]?.url || '/placeholder.png'}
   180→   180→                alt={images[idx]?.alt || 'Product image'}
   181→   181→                fill
   182→   182→                className="object-contain"
   183→   183→                sizes="90vw"
   184→   184→                priority
   185→   185→              />
   186→   186→            </motion.div>
   187→   187→          </AnimatePresence>
   188→   188→          {/* Close */}
   189→   189→          <button
   190→   190→            onClick={onClose}
   191→   191→            className="absolute top-4 right-4 z-10 h-10 w-10 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors"
   192→   192→            aria-label="Close viewer"
   193→   193→          >
   194→   194→            <X className="h-5 w-5" />
   195→   195→          </button>
   196→   196→          {/* Nav arrows */}
   197→   197→          {images.length > 1 && (
   198→   198→            <>
   199→   199→              <button
   200→   200→                onClick={() => setIdx((i) => Math.max(0, i - 1))}
   201→   201→                disabled={idx === 0}
   202→   202→                className="absolute left-4 top-1/2 -translate-y-1/2 z-10 h-12 w-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors disabled:opacity-30"
   203→   203→                aria-label="Previous image"
   204→   204→              >
   205→   205→                <ChevronLeft className="h-6 w-6" />
   206→   206→              </button>
   207→   207→              <button
   208→   208→                onClick={() => setIdx((i) => Math.min(images.length - 1, i + 1))}
   209→   209→                disabled={idx === images.length - 1}
   210→   210→                className="absolute right-4 top-1/2 -translate-y-1/2 z-10 h-12 w-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center text-white hover:bg-white/30 transition-colors disabled:opacity-30"
   211→   211→                aria-label="Next image"
   212→   212→              >
   213→   213→                <ChevronRight className="h-6 w-6" />
   214→   214→              </button>
   215→   215→            </>
   216→   216→          )}
   217→   217→          {/* Counter */}
   218→   218→          <div className="absolute bottom-4 left-1/2 -translate-x-1/2 z-10 bg-black/60 backdrop-blur-sm text-white text-sm px-3 py-1.5 rounded-full">
   219→   219→            {idx + 1} / {images.length}
   220→   220→          </div>
   221→   221→          {/* Thumbnails */}
   222→   222→          {images.length > 1 && (
   223→   223→            <div className="absolute bottom-14 left-1/2 -translate-x-1/2 z-10 flex gap-2 max-w-md overflow-x-auto px-2 py-1 scrollbar-hide">
   224→   224→              {images.map((img, i) => (
   225→   225→                <button
   226→   226→                  key={img.id}
   227→   227→                  onClick={() => setIdx(i)}
   228→   228→                  className={`relative h-12 w-12 rounded-md overflow-hidden flex-shrink-0 border-2 transition-all ${
   229→   229→                    i === idx ? 'border-white ring-2 ring-white/30' : 'border-white/30 opacity-60 hover:opacity-90'
   230→   230→                  }`}
   231→   231→                >
   232→   232→                  <Image src={img.url} alt={img.alt || ''} fill className="object-cover" sizes="48px" />
   233→   233→                </button>
   234→   234→              ))}
   235→   235→            </div>
   236→   236→          )}
   237→   237→        </div>
   238→   238→      </DialogContent>
   239→   239→    </Dialog>
   240→   240→  )
   241→   241→}

   245→   245→function ScrollToTopButton() {
   246→   246→  const [visible, setVisible] = useState(false)
   247→   247→
   248→   248→  useEffect(() => {
   249→   249→    const handleScroll = () => setVisible(window.scrollY > 600)
   250→   250→    window.addEventListener('scroll', handleScroll, { passive: true })
   251→   251→    return () => window.removeEventListener('scroll', handleScroll)
   252→   252→  }, [])
   253→   253→
   254→   254→  return (
   255→   255→    <AnimatePresence>
   256→   256→      {visible && (
   257→   257→        <motion.button
   258→   258→          initial={{ opacity: 0, scale: 0.8 }}
   259→   259→          animate={{ opacity: 1, scale: 1 }}
   260→   260→          exit={{ opacity: 0, scale: 0.8 }}
   261→   261→          onClick={() => window.scrollTo({ top: 0, behavior: 'smooth' })}
   262→   262→          className="fixed bottom-20 md:bottom-8 right-4 z-40 h-10 w-10 rounded-full bg-emerald-600 text-white shadow-lg shadow-emerald-600/30 flex items-center justify-center hover:bg-emerald-700 transition-colors"
   263→   263→          aria-label="Scroll to top"
   264→   264→        >
   265→   265→          <ArrowUp className="h-4 w-4" />
   266→   266→        </motion.button>
   267→   267→      )}
   268→   268→    </AnimatePresence>
   269→   269→  )
   270→   270→}
