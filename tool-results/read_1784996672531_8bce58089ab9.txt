    73→  unit: string

    75→  categoryId: string
    76→  subcategoryId?: string | null
    77→  origin?: string
    78→  qualityNote?: string
    79→  ingredients?: string
    80→  storageInstruction?: string
    81→  deliveryNote?: string
    82→  status: string
    83→  isFeatured: boolean
    84→  isBestSeller: boolean
    85→  isNewArrival: boolean
    86→  tags?: string[] | string
    87→  specifications?: Record<string, string> | string
    88→  images?: Array<{ id: string; url: string; alt?: string }>
    89→  videoUrl?: string
    90→  metaTitle?: string
    91→  metaDescription?: string
    92→  metaKeywords?: string
    93→  category?: { id: string; name: string }

   100→  subcategories?: Array<{ id: string; name: string }>

   104→  name: '',
   105→  slug: '',
   106→  description: '',
   107→  shortDescription: '',
   108→  price: 0,
   109→  discountPrice: null,
   110→  unit: 'kg',
   111→  stock: 0,
   112→  categoryId: '',
   113→  subcategoryId: '',
   114→  origin: '',
   115→  qualityNote: '',
   116→  ingredients: '',
   117→  storageInstruction: '',
   118→  deliveryNote: '',
   119→  status: 'active',
   120→  isFeatured: false,
   121→  isBestSeller: false,
   122→  isNewArrival: false,
   123→  tags: [],
   124→  specifications: {},
   125→  images: [],
   126→  videoUrl: '',
   127→  metaTitle: '',
   128→  metaDescription: '',
   129→  metaKeywords: '',

   132→export default function AdminProductsPage() {
   133→  const [products, setProducts] = useState<Product[]>([])
   134→  const [categories, setCategories] = useState<Category[]>([])
   135→  const [loading, setLoading] = useState(true)
   136→  const [search, setSearch] = useState('')
   137→  const [statusFilter, setStatusFilter] = useState('all')
   138→  const [categoryFilter, setCategoryFilter] = useState('all')
   139→  const [dialogOpen, setDialogOpen] = useState(false)
   140→  const [editingProduct, setEditingProduct] = useState<Partial<Product> | null>(null)
   141→  const [deleteId, setDeleteId] = useState<string | null>(null)
   142→  const [saving, setSaving] = useState(false)
   143→  const [selectedIds, setSelectedIds] = useState<string[]>([])
   144→  const [tagInput, setTagInput] = useState('')
   145→  const [specKey, setSpecKey] = useState('')
   146→  const [specValue, setSpecValue] = useState('')
   147→  const [activeTab, setActiveTab] = useState('basic')
   148→
   149→  const fetchProducts = useCallback(async () => {
   150→    try {
   151→      const params = new URLSearchParams()
   152→      if (search) params.set('search', search)
   153→      if (statusFilter !== 'all') params.set('status', statusFilter)
   154→      if (categoryFilter !== 'all') params.set('categoryId', categoryFilter)
   155→      params.set('limit', '50')
   156→      const res = await fetch(`/api/products?${params.toString()}`)
   157→      if (res.ok) {
   158→        const data = await res.json()
   159→        setProducts(extractItems(data))
   160→      }
   161→    } catch {
   162→      toast.error('Failed to fetch products')
   163→    } finally {
   164→      setLoading(false)
   165→    }
   166→  }, [search, statusFilter, categoryFilter])
   167→
   168→  const fetchCategories = useCallback(async () => {
   169→    try {
   170→      const res = await fetch('/api/categories')
   171→      if (res.ok) {
   172→        const data = await res.json()
   173→        setCategories(extractItems(data))
   174→      }
   175→    } catch {
   176→      // silent
   177→    }
   178→  }, [])
   179→
   180→  useEffect(() => {
   181→    fetchCategories()
   182→  }, [fetchCategories])
   183→
   184→  useEffect(() => {
   185→    fetchProducts()
   186→  }, [fetchProducts])
   187→
   188→  function generateSlug(name: string): string {
   189→    return name
   190→      .toLowerCase()
   191→      .replace(/[^a-z0-9\s-]/g, '')
   192→      .replace(/\s+/g, '-')
   193→      .replace(/-+/g, '-')
   194→      .trim()
   195→  }
   196→
   197→  function openCreate() {
   198→    setEditingProduct({ ...emptyProduct })
   199→    setActiveTab('basic')
   200→    setDialogOpen(true)
   201→  }
   202→
   203→  function openEdit(product: Product) {
   204→    setEditingProduct({
   205→      ...product,
   206→      tags: Array.isArray(product.tags)
   207→        ? product.tags
   208→        : typeof product.tags === 'string'
   209→          ? product.tags.split(',').map((tag) => tag.trim()).filter(Boolean)
   210→          : [],
   211→      specifications:
   212→        typeof product.specifications === 'string'
   213→          ? JSON.parse(product.specifications || '{}')
   214→          : product.specifications ?? {},
   215→    })
   216→    setActiveTab('basic')
   217→    setDialogOpen(true)
   218→  }
   219→
   220→  async function handleSave() {
   221→    if (!editingProduct?.name) {
   222→      toast.error('Product name is required')
   223→      return
   224→    }
   225→    setSaving(true)
   226→    try {
   227→      const payload = {
   228→        ...editingProduct,
   229→        slug: editingProduct.slug || generateSlug(editingProduct.name),
   230→        price: Number(editingProduct.price) || 0,
   231→        discountPrice: editingProduct.discountPrice ? Number(editingProduct.discountPrice) : null,
   232→        stock: Number(editingProduct.stock) || 0,
   233→      }
   234→
   235→      const isEdit = !!editingProduct.id
   236→      const url = isEdit ? `/api/products/${editingProduct.id}` : '/api/products'
   237→      const method = isEdit ? 'PUT' : 'POST'
   238→
   239→      const res = await fetch(url, {
   240→        method,
   241→        headers: { 'Content-Type': 'application/json' },
   242→        body: JSON.stringify(payload),
   243→      })
   244→
   245→      if (!res.ok) {
   246→        const data = await res.json()
   247→        throw new Error(data.error || 'Failed to save product')
   248→      }
   249→
   250→      toast.success(isEdit ? 'Product updated' : 'Product created')
   251→      setDialogOpen(false)
   252→      fetchProducts()
   253→    } catch (err) {
   254→      toast.error(err instanceof Error ? err.message : 'Failed to save product')
   255→    } finally {
   256→      setSaving(false)
   257→    }
   258→  }
   259→
   260→  async function handleDelete(id: string) {
   261→    try {
   262→      const res = await fetch(`/api/products/${id}`, { method: 'DELETE' })
   263→      if (!res.ok) throw new Error('Failed to delete')
   264→      toast.success('Product deleted')
   265→      fetchProducts()
   266→    } catch {
   267→      toast.error('Failed to delete product')
   268→    } finally {
   269→      setDeleteId(null)
   270→    }
   271→  }
   272→
   273→  async function handleBulkAction(action: string) {
   274→    if (selectedIds.length === 0) return
   275→    try {
   276→      await Promise.all(
   277→        selectedIds.map((id) =>
   278→          fetch(`/api/products/${id}`, {
   279→            method: 'PUT',
   280→            headers: { 'Content-Type': 'application/json' },
   281→            body: JSON.stringify({ status: action }),
   282→          })
   283→        )
   284→      )
   285→      toast.success(`${selectedIds.length} products updated`)
   286→      setSelectedIds([])
   287→      fetchProducts()
   288→    } catch {
   289→      toast.error('Bulk action failed')
   290→    }
   291→  }
   292→
   293→  function toggleSelect(id: string) {
   294→    setSelectedIds((prev) =>
   295→      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id]
   296→    )
   297→  }
   298→
   299→  function addTag() {
   300→    if (!tagInput.trim()) return
   301→    const currentTags = Array.isArray(editingProduct?.tags) ? editingProduct.tags : []
   302→    const tags = [...currentTags, tagInput.trim()]
   303→    setEditingProduct((prev) => ({ ...prev, tags }) as Partial<Product>)
   304→    setTagInput('')
   305→  }
   306→
   307→  function removeTag(index: number) {
   308→    const tags = Array.isArray(editingProduct?.tags)
   309→      ? editingProduct.tags.filter((_, i) => i !== index)
   310→      : []
   311→    setEditingProduct((prev) => ({ ...prev, tags }) as Partial<Product>)
   312→  }
   313→
   314→  function addSpec() {
   315→    if (!specKey.trim()) return
   316→    const currentSpecifications =
   317→      editingProduct?.specifications && typeof editingProduct.specifications !== 'string'
   318→        ? editingProduct.specifications
   319→        : {}
   320→    const specifications = { ...currentSpecifications, [specKey.trim()]: specValue.trim() }
   321→    setEditingProduct((prev) => ({ ...prev, specifications }) as Partial<Product>)
   322→    setSpecKey('')
   323→    setSpecValue('')
   324→  }
   325→
   326→  function removeSpec(key: string) {
   327→    const specifications =
   328→      editingProduct?.specifications && typeof editingProduct.specifications !== 'string'
   329→        ? { ...editingProduct.specifications }
   330→        : {}
   331→    delete specifications[key]
   332→    setEditingProduct((prev) => ({ ...prev, specifications }) as Partial<Product>)
   333→  }
   334→
   335→  async function handleImageUpload(e: React.ChangeEvent<HTMLInputElement>) {
   336→    const files = e.target.files
   337→    if (!files) return
   338→    const formData = new FormData()
   339→    Array.from(files).forEach((f) => formData.append('files', f))
   340→    try {
   341→      const res = await fetch('/api/uploads', { method: 'POST', body: formData })
   342→      if (!res.ok) throw new Error('Upload failed')
   343→      const data = await res.json()
   344→      const newImages = (data.files ?? []).map((f: { url: string; id?: string }) => ({
   345→        id: f.id ?? `img_${Date.now()}`,
   346→        url: f.url,
   347→        alt: '',
   348→      }))
   349→      setEditingProduct((prev) => ({
   350→        ...prev,
   351→        images: [...(prev?.images ?? []), ...newImages],
   352→      }) as Partial<Product>)
   353→    } catch {
   354→      toast.error('Image upload failed')
   355→    }
   356→  }
   357→
   358→  function removeImage(index: number) {
   359→    const images = editingProduct?.images?.filter((_, i) => i !== index) ?? []
   360→    setEditingProduct((prev) => ({ ...prev, images }) as Partial<Product>)
   361→  }
   362→
   363→  const selectedCategory = categories.find((c) => c.id === editingProduct?.categoryId)
   364→
   365→  return (
   366→    <div className="space-y-4">
   367→      {/* Filters */}
   368→      <Card>
   369→        <CardContent className="p-4">
   370→          <div className="flex flex-col sm:flex-row gap-3">
   371→            <div className="relative flex-1">
   372→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   373→              <Input
   374→                placeholder="Search products..."
   375→                value={search}
   376→                onChange={(e) => setSearch(e.target.value)}
   377→                className="pl-9"
   378→              />
   379→            </div>
   380→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   381→              <SelectTrigger className="w-full sm:w-40">
   382→                <SelectValue placeholder="Status" />
   383→              </SelectTrigger>
   384→              <SelectContent>
   385→                <SelectItem value="all">All Status</SelectItem>
   386→                <SelectItem value="active">Active</SelectItem>
   387→                <SelectItem value="inactive">Inactive</SelectItem>
   388→                <SelectItem value="archived">Archived</SelectItem>
   389→              </SelectContent>
   390→            </Select>
   391→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   392→              <SelectTrigger className="w-full sm:w-48">
   393→                <SelectValue placeholder="Category" />
   394→              </SelectTrigger>
   395→              <SelectContent>
   396→                <SelectItem value="all">All Categories</SelectItem>
   397→                {categories.filter((cat) => cat.id).map((cat) => (
   398→                  <SelectItem key={cat.id} value={cat.id}>
   399→                    {cat.name}
   400→                  </SelectItem>
   401→                ))}
   402→              </SelectContent>
   403→            </Select>
   404→            <Button onClick={openCreate} className="gap-2">
   405→              <Plus className="size-4" /> Add Product
   406→            </Button>
   407→          </div>
   408→        </CardContent>
   409→      </Card>
   410→
   411→      {/* Bulk Actions */}
   412→      {selectedIds.length > 0 && (
   413→        <Card>
   414→          <CardContent className="p-3">
   415→            <div className="flex items-center gap-3">
   416→              <span className="text-sm text-muted-foreground">
   417→                {selectedIds.length} selected
   418→              </span>
   419→              <Button size="sm" variant="outline" onClick={() => handleBulkAction('active')}>
   420→                Set Active
   421→              </Button>
   422→              <Button size="sm" variant="outline" onClick={() => handleBulkAction('inactive')}>
   423→                Set Inactive
   424→              </Button>
   425→              <Button
   426→                size="sm"
   427→                variant="ghost"
   428→                onClick={() => setSelectedIds([])}
   429→              >
   430→                Clear
   431→              </Button>
   432→            </div>
   433→          </CardContent>
   434→        </Card>
   435→      )}
   436→
   437→      {/* Products Table */}
   438→      <Card>
   439→        <CardContent className="p-0">
   440→          {loading ? (
   441→            <div className="p-6 space-y-3">
   442→              {[...Array(5)].map((_, i) => (
   443→                <Skeleton key={i} className="h-12 w-full" />
   444→              ))}
   445→            </div>
   446→          ) : (
   447→            <Table>
   448→              <TableHeader>
   449→                <TableRow>
   450→                  <TableHead className="w-10">
   451→                    <Checkbox
   452→                      checked={selectedIds.length === products.length && products.length > 0}
   453→                      onCheckedChange={(checked) => {
   454→                        if (checked) {
   455→                          setSelectedIds(products.map((p) => p.id))
   456→                        } else {
   457→                          setSelectedIds([])
   458→                        }
   459→                      }}
   460→                    />
   461→                  </TableHead>
   462→                  <TableHead>Product</TableHead>
   463→                  <TableHead>Category</TableHead>
   464→                  <TableHead>Price</TableHead>
   465→                  <TableHead>Stock</TableHead>
   466→                  <TableHead>Status</TableHead>
   467→                  <TableHead>Flags</TableHead>
   468→                  <TableHead className="w-10" />
   469→                </TableRow>
   470→              </TableHeader>
   471→              <TableBody>
   472→                {products.length === 0 ? (
   473→                  <TableRow>
   474→                    <TableCell colSpan={8} className="text-center py-12">
   475→                      <Package className="mx-auto size-10 text-muted-foreground mb-2" />
   476→                      <p className="text-muted-foreground">No products found</p>
   477→                      <Button variant="outline" size="sm" className="mt-2" onClick={openCreate}>
   478→                        Add your first product
   479→                      </Button>
   480→                    </TableCell>
   481→                  </TableRow>
   482→                ) : (
   483→                  products.map((product) => (
   484→                    <TableRow key={product.id}>
   485→                      <TableCell>
   486→                        <Checkbox
   487→                          checked={selectedIds.includes(product.id)}
   488→                          onCheckedChange={() => toggleSelect(product.id)}
   489→                        />
   490→                      </TableCell>
   491→                      <TableCell>
   492→                        <div className="flex items-center gap-3">
   493→                          {product.images?.[0]?.url ? (
   494→                            <img
   495→                              src={product.images[0].url}
   496→                              alt={product.name}
   497→                              className="size-10 rounded-md object-cover"
   498→                            />
   499→                          ) : (
   500→                            <div className="size-10 rounded-md bg-muted flex items-center justify-center">
   501→                              <Package className="size-4 text-muted-foreground" />
   502→                            </div>
   503→                          )}
   504→                          <div>
   505→                            <p className="font-medium text-sm">{product.name}</p>
   506→                            <p className="text-xs text-muted-foreground">{product.slug}</p>
   507→                          </div>
   508→                        </div>
   509→                      </TableCell>
   510→                      <TableCell className="text-sm">
   511→                        {product.category?.name ?? '-'}
   512→                      </TableCell>
   513→                      <TableCell>
   514→                        <div className="text-sm">
   515→                          <span className="font-medium">৳{product.price}</span>
   516→                          {product.discountPrice && (
   517→                            <span className="ml-1 text-xs text-muted-foreground line-through">
   518→                              ৳{product.discountPrice}
   519→                            </span>
   520→                          )}
   521→                        </div>
   522→                      </TableCell>
   523→                      <TableCell>
   524→                        <Badge
   525→                          variant={product.stock <= 5 ? 'destructive' : 'secondary'}
   526→                          className="text-xs"
   527→                        >
   528→                          {product.stock} {product.unit}
   529→                        </Badge>
   530→                      </TableCell>
   531→                      <TableCell>
   532→                        <Badge
   533→                          variant="secondary"
   534→                          className={
   535→                            product.status === 'active'
   536→                              ? 'bg-green-100 text-green-800'
   537→                              : product.status === 'inactive'
   538→                              ? 'bg-gray-100 text-gray-800'
   539→                              : 'bg-yellow-100 text-yellow-800'
   540→                          }
   541→                        >
   542→                          {product.status}
   543→                        </Badge>
   544→                      </TableCell>
   545→                      <TableCell>
   546→                        <div className="flex gap-1">
   547→                          {product.isFeatured && (
   548→                            <Badge variant="secondary" className="text-xs bg-blue-50 text-blue-700">
   549→                              Featured
   550→                            </Badge>
   551→                          )}
   552→                          {product.isBestSeller && (
   553→                            <Badge variant="secondary" className="text-xs bg-orange-50 text-orange-700">
   554→                              Best
   555→                            </Badge>
   556→                          )}
   557→                          {product.isNewArrival && (
   558→                            <Badge variant="secondary" className="text-xs bg-purple-50 text-purple-700">
   559→                              New
   560→                            </Badge>
   561→                          )}
   562→                        </div>
   563→                      </TableCell>
   564→                      <TableCell>
   565→                        <DropdownMenu>
   566→                          <DropdownMenuTrigger asChild>
   567→                            <Button variant="ghost" size="icon" className="size-8">
   568→                              <MoreHorizontal className="size-4" />
   569→                            </Button>
   570→                          </DropdownMenuTrigger>
   571→                          <DropdownMenuContent align="end">
   572→                            <DropdownMenuItem onClick={() => openEdit(product)}>
   573→                              <Pencil className="mr-2 size-4" /> Edit
   574→                            </DropdownMenuItem>
   575→                            <DropdownMenuItem
   576→                              className="text-destructive"
   577→                              onClick={() => setDeleteId(product.id)}
   578→                            >
   579→                              <Trash2 className="mr-2 size-4" /> Delete
   580→                            </DropdownMenuItem>
   581→                          </DropdownMenuContent>
   582→                        </DropdownMenu>
   583→                      </TableCell>
   584→                    </TableRow>
   585→                  ))
   586→                )}
   587→              </TableBody>
   588→            </Table>
   589→          )}
   590→        </CardContent>
   591→      </Card>
   592→
   593→      {/* Add/Edit Dialog */}
   594→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   595→        <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   596→          <DialogHeader>
   597→            <DialogTitle>
   598→              {editingProduct?.id ? 'Edit Product' : 'Add New Product'}
   599→            </DialogTitle>
   600→          </DialogHeader>
   601→
   602→          <Tabs value={activeTab} onValueChange={setActiveTab}>
   603→            <TabsList className="w-full flex-wrap">
   604→              <TabsTrigger value="basic">Basic Info</TabsTrigger>
   605→              <TabsTrigger value="pricing">Pricing & Stock</TabsTrigger>
   606→              <TabsTrigger value="details">Details</TabsTrigger>
   607→              <TabsTrigger value="images">Images</TabsTrigger>
   608→              <TabsTrigger value="seo">SEO</TabsTrigger>
   609→            </TabsList>
   610→
   611→            <TabsContent value="basic" className="space-y-4 mt-4">
   612→              <div className="grid gap-4 sm:grid-cols-2">
   613→                <div className="space-y-2">
   614→                  <Label>Product Name *</Label>
   615→                  <Input
   616→                    value={editingProduct?.name ?? ''}
   617→                    onChange={(e) => {
   618→                      const name = e.target.value
   619→                      setEditingProduct((prev) => ({
   620→                        ...prev,
   621→                        name,
   622→                        slug: prev?.id ? prev.slug : generateSlug(name),
   623→                      }) as Partial<Product>)
