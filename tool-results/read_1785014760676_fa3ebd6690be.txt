   134→function getStepIndex(status: string): number {
   135→  if (status === 'cancelled') return -1
   136→  const idx = statusSteps.findIndex((s) => s.key === status)
   137→  return idx >= 0 ? idx : 0
   138→}

   140→function formatPaymentMethod(method: string): string {
   141→  const map: Record<string, string> = {
   142→    cod: 'Cash on Delivery',
   143→    online: 'SSLCommerz Payment Gateway',
   144→    card: 'Card Payment',
   145→  }
   146→  return map[method] || method.charAt(0).toUpperCase() + method.slice(1)
   147→}

   151→function VerticalTimeline({ currentStatus, createdAt }: { currentStatus: string; createdAt: string }) {
   152→  const currentStep = getStepIndex(currentStatus)
   153→  const isCancelled = currentStatus === 'cancelled'
   154→
   155→  if (isCancelled) {
   156→    return (
   157→      <div className="relative pl-8">
   158→        <div className="absolute left-[15px] top-2 bottom-2 w-0.5 bg-red-200" />
   159→        <div className="relative flex items-start gap-4 pb-5">
   160→          <div className="absolute left-[-21px] h-8 w-8 rounded-full bg-gradient-to-br from-red-400 to-red-600 text-white flex items-center justify-center z-10 shadow-md shadow-red-200">
   161→            <X className="h-4 w-4" />
   162→          </div>
   163→          <div className="flex-1 ml-2">
   164→            <span className="text-sm font-bold text-red-700">Order Cancelled</span>
   165→            <p className="text-xs text-red-500 mt-0.5">This order has been cancelled</p>
   166→          </div>
   167→        </div>
   168→      </div>
   169→    )
   170→  }
   171→
   172→  return (
   173→    <div className="relative pl-8">
   174→      {/* Vertical line */}
   175→      <div className="absolute left-[15px] top-2 bottom-2 w-0.5 bg-gradient-to-b from-emerald-400 to-emerald-200" />
   176→
   177→      {statusSteps.map((step, idx) => {
   178→        const StepIcon = step.icon
   179→        const isCompleted = idx <= currentStep
   180→        const isCurrent = idx === currentStep
   181→
   182→        return (
   183→          <motion.div
   184→            key={step.key}
   185→            initial={{ opacity: 0, x: -10 }}
   186→            animate={{ opacity: 1, x: 0 }}
   187→            transition={{ delay: idx * 0.1 }}
   188→            className="relative flex items-start gap-4 pb-6 last:pb-0"
   189→          >
   190→            <div className={`absolute left-[-21px] h-8 w-8 rounded-full flex items-center justify-center z-10 ${
   191→              isCompleted
   192→                ? 'bg-gradient-to-br from-emerald-500 to-emerald-700 text-white shadow-md shadow-emerald-200'
   193→                : isCurrent
   194→                ? 'bg-emerald-100 text-emerald-600 border-2 border-emerald-400 shadow-md'
   195→                : 'bg-gray-50 text-gray-400 border-2 border-gray-200'
   196→            }`}>
   197→              {isCompleted ? (
   198→                <CheckCircle2 className="h-4 w-4" />
   199→              ) : (
   200→                <StepIcon className="h-4 w-4" />
   201→              )}
   202→            </div>
   203→            <div className="flex-1 ml-2">
   204→              <div className="flex items-center gap-2">
   205→                <span className={`text-sm font-bold ${isCompleted ? 'text-emerald-700' : isCurrent ? 'text-gray-800' : 'text-gray-400'}`}>
   206→                  {step.label}
   207→                </span>
   208→                {isCompleted && idx < statusSteps.length - 1 && (
   209→                  <Badge className="bg-emerald-100 text-emerald-700 border-0 text-[10px] h-5 px-2 font-semibold">Done</Badge>
   210→                )}
   211→                {isCurrent && (
   212→                  <Badge className="bg-amber-100 text-amber-700 border-0 text-[10px] h-5 px-2 font-semibold">Current</Badge>
   213→                )}
   214→              </div>
   215→              <p className={`text-xs mt-1 ${isCompleted || isCurrent ? 'text-gray-500' : 'text-gray-400'}`}>
   216→                {step.desc}
   217→              </p>
   218→              {idx === 0 && (
   219→                <p className="text-xs text-gray-400 mt-1 flex items-center gap-1">
   220→                  <Clock className="h-3 w-3" />
   221→                  {new Date(createdAt).toLocaleDateString('en-BD', { month: 'short', day: 'numeric', year: 'numeric', hour: '2-digit', minute: '2-digit' })}
   222→                </p>
   223→              )}
   224→            </div>
   225→          </motion.div>
   226→        )
   227→      })}
   228→    </div>
   229→  )
   230→}
