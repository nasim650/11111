   135→   135→// ── Flash Sale Countdown ──
   136→   136→function FlashSaleCountdown() {
   137→   137→  const [timeLeft, setTimeLeft] = useState({ hours: 0, minutes: 0, seconds: 0 })
   138→   138→
   139→   139→  useEffect(() => {
   140→   140→    const getTargetTime = () => {
   141→   141→      const now = new Date()
   142→   142→      const end = new Date(now)
   143→   143→      end.setHours(23, 59, 59, 999)
   144→   144→      return end.getTime()
   145→   145→    }
   146→   146→
   147→   147→    const updateCountdown = () => {
   148→   148→      const target = getTargetTime()
   149→   149→      const now = Date.now()
   150→   150→      const diff = Math.max(target - now, 0)
   151→   151→
   152→   152→      setTimeLeft({
   153→   153→        hours: Math.floor(diff / (1000 * 60 * 60)),
   154→   154→        minutes: Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)),
   155→   155→        seconds: Math.floor((diff % (1000 * 60)) / 1000),
   156→   156→      })
   157→   157→    }
   158→   158→
   159→   159→    updateCountdown()
   160→   160→    const timer = setInterval(updateCountdown, 1000)
   161→   161→    return () => clearInterval(timer)
   162→   162→  }, [])
   163→   163→
   164→   164→  const pad = (n: number) => n.toString().padStart(2, '0')
   165→   165→  const isEndingSoon = timeLeft.hours === 0 && timeLeft.minutes < 60
   166→   166→
   167→   167→  return (
   168→   168→    <div className="flex items-center gap-2">
   169→   169→      {isEndingSoon && (
   170→   170→        <span
   171→   171→          className="text-[10px] font-medium bg-white/15 text-white px-2 py-0.5 rounded-full mr-1"
   172→   172→        >
   173→   173→          Ending soon
   174→   174→        </span>
   175→   175→      )}
   176→   176→      <div className="flex items-center gap-1">
   177→   177→        <FlipDigit value={pad(timeLeft.hours)} label="ঘণ্টা" />
   178→   178→        <span className="text-white font-bold text-xl mx-0.5 animate-pulse">:</span>
   179→   179→        <FlipDigit value={pad(timeLeft.minutes)} label="মিনিট" />
   180→   180→        <span className="text-white font-bold text-xl mx-0.5 animate-pulse">:</span>
   181→   181→        <FlipDigit value={pad(timeLeft.seconds)} label="সেকেন্ড" />
   182→   182→      </div>
   183→   183→    </div>
   184→   184→  )
   185→   185→}

   262→   262→function HeroBannerCarousel({ banners, navigate }: {
   263→   263→  banners: Banner[]
   264→   264→  navigate: (page: PageName, params?: Record<string, unknown>) => void
   265→   265→}) {
   266→   266→  const [currentBanner, setCurrentBanner] = useState(0)
   267→   267→  const [isPaused, setIsPaused] = useState(false)
   268→   268→  const [progress, setProgress] = useState(0)
   269→   269→  const [scrollY, setScrollY] = useState(0)
   270→   270→  const intervalRef = useRef<NodeJS.Timeout | null>(null)
   271→   271→
   272→   272→  // Parallax scroll tracking
   273→   273→  useEffect(() => {
   274→   274→    const handleScroll = () => {
   275→   275→      setScrollY(window.scrollY)
   276→   276→    }
   277→   277→    window.addEventListener('scroll', handleScroll, { passive: true })
   278→   278→    return () => window.removeEventListener('scroll', handleScroll)
   279→   279→  }, [])
   280→   280→
   281→   281→  // Auto-play with progress
   282→   282→  useEffect(() => {
   283→   283→    if (banners.length <= 1 || isPaused) return
   284→   284→
   285→   285→    const SLIDE_DURATION = 5000
   286→   286→    const TICK = 50
   287→   287→    let elapsed = 0
   288→   288→
   289→   289→    intervalRef.current = setInterval(() => {
   290→   290→      elapsed += TICK
   291→   291→      setProgress((elapsed / SLIDE_DURATION) * 100)
   292→   292→
   293→   293→      if (elapsed >= SLIDE_DURATION) {
   294→   294→        elapsed = 0
   295→   295→        setCurrentBanner((prev) => (prev + 1) % banners.length)
   296→   296→        setProgress(0)
   297→   297→      }
   298→   298→    }, TICK)
   299→   299→
   300→   300→    return () => {
   301→   301→      if (intervalRef.current) clearInterval(intervalRef.current)
   302→   302→    }
   303→   303→  }, [banners.length, isPaused, currentBanner])
   304→   304→
   305→   305→  const handlePrev = () => {
   306→   306→    setCurrentBanner((prev) => (prev - 1 + banners.length) % banners.length)
   307→   307→    setProgress(0)
   308→   308→  }
   309→   309→
   310→   310→  const handleNext = () => {
   311→   311→    setCurrentBanner((prev) => (prev + 1) % banners.length)
   312→   312→    setProgress(0)
   313→   313→  }
   314→   314→
   315→   315→  const handleDotClick = (idx: number) => {
   316→   316→    setCurrentBanner(idx)
   317→   317→    setProgress(0)
   318→   318→  }
   319→   319→
   320→   320→  if (banners.length === 0) return null
   321→   321→
   322→   322→  // Parallax offset - capped at 60px
   323→   323→  const parallaxOffset = Math.min(scrollY * 0.3, 60)
   324→   324→
   325→   325→  return (
   326→   326→    <div
   327→   327→      className="relative h-64 sm:h-80 md:h-[440px] lg:h-[520px] overflow-hidden bg-muted"
   328→   328→      onMouseEnter={() => setIsPaused(true)}

   331→   331→      <AnimatePresence mode="wait">
   332→   332→        <motion.div
   333→   333→          key={currentBanner}
   334→   334→          initial={{ opacity: 0, x: 60 }}

   336→   336→          exit={{ opacity: 0, x: -60 }}

   338→   338→          className="absolute inset-0"
   339→   339→        >
   340→   340→          {banners[currentBanner].desktopImage ? (
   341→   341→            <>
   342→   342→              {/* Parallax image container */}
   343→   343→              <div
   344→   344→                className="absolute inset-0 w-full h-full"
   345→   345→                style={{ transform: `translateY(${parallaxOffset}px) scale(1.1)` }}
   346→   346→              >

   349→   349→                  alt={banners[currentBanner].title}
